overtake.plots

Figures for the project documentation, written as PNGs.

Each public function draws one figure from results computed elsewhere, mostly the dictionaries that overtake.experiments saves as JSON in reports/results/, and writes it to path. overtake.experiments and overtake.train call them as they produce results, and overtake.experiments.figures redraws the figures from the saved JSON without rerunning anything.

Importing the module selects matplotlib's non-interactive Agg backend and sets the house style in matplotlib.pyplot.rcParams for the whole process.

Colours follow a colour-vision-deficiency-checked palette: up to three categorical series (blue, orange, green) and a single-hue blue ramp for ordered values such as battery charge. Every multi-series figure has a legend.

 81def _label_points(ax, xs, ys, labels, marker_pt=4):
 82    """Label scatter points without labels overlapping each other, the points or the axes edge.
 83
 84    Greedy placement: the most crowded points are labelled first, while they
 85    still have free space. Each label tries ten positions around its point (to
 86    the right, above, below or left, then further above or below) and takes the
 87    first that stays inside the axes and clears every point, every label placed
 88    so far and any text already in the axes. If none is free it takes the first
 89    position (right, slightly above) anyway. Boxes are measured in display
 90    pixels, so call this after the layout is final (``fig.tight_layout()`` and
 91    ``fig.canvas.draw()``).
 92
 93    Parameters
 94    ----------
 95    ax : matplotlib.axes.Axes
 96    xs, ys : sequence of float
 97        Point positions in data coordinates.
 98    labels : sequence of str
 99        One label per point.
100    marker_pt : float, optional
101        Half-width, in points, of the square kept clear around each point
102        (about the marker radius).
103    """
104    fig = ax.figure
105    renderer = fig.canvas.get_renderer()
106    radius = marker_pt * fig.dpi / 72  # points -> display pixels (72 points per inch)
107    points = ax.transData.transform(np.column_stack([xs, ys]))
108    # Boxes no label may overlap: every marker, then any text already in the axes (e.g. a statistics note).
109    taken = [matplotlib.transforms.Bbox.from_extents(px - radius, py - radius, px + radius, py + radius)
110             for px, py in points]
111    taken += [t.get_window_extent(renderer) for t in ax.texts]
112    # Crowding: points within 60 pixels, the point itself included (threshold: estimate).
113    near = [(np.hypot(*(points - pt).T) < 60).sum() for pt in points]
114    # (offset in points, horizontal alignment, vertical alignment), in order of preference.
115    candidates = [((5, 2), "left", "bottom"), ((5, -2), "left", "top"), ((0, 6), "center", "bottom"),
116                  ((0, -6), "center", "top"), ((-5, 2), "right", "bottom"), ((-5, -2), "right", "top"),
117                  ((6, 0), "left", "center"), ((-6, 0), "right", "center"),
118                  ((0, 14), "center", "bottom"), ((0, -14), "center", "top")]
119
120    def place(i, offset, ha, va):
121        """Annotate point ``i`` at ``offset`` points; return the text and its box padded by 1 pixel."""
122        text = ax.annotate(labels[i], (xs[i], ys[i]), xytext=offset, textcoords="offset points", ha=ha, va=va,
123                           color=INK_SECONDARY, fontsize=10)
124        return text, text.get_window_extent(renderer).padded(1)
125
126    for i in sorted(range(len(labels)), key=lambda i: -near[i]):
127        for candidate in candidates:
128            text, box = place(i, *candidate)
129            inside = ax.bbox.contains(box.x0, box.y0) and ax.bbox.contains(box.x1, box.y1)
130            if inside and not any(box.overlaps(t) for t in taken):
131                break
132            text.remove()
133        else:  # nowhere free: fall back to the first position
134            text, box = place(i, *candidates[0])
135        taken.append(box)

168def calibration(y, curves, path, bins=10):
169    """Reliability diagram: predicted probability against observed overtake rate.
170
171    Predictions are grouped into ``bins`` equal-width probability bins; each
172    point is a bin's mean prediction against the share of its attempts that
173    were overtakes. Bins with fewer than 20 attempts are left out.
174
175    Parameters
176    ----------
177    y : numpy.ndarray
178        Observed 0/1 outcomes.
179    curves : dict of str -> numpy.ndarray
180        Model name -> predicted probabilities for the same attempts as ``y``.
181        At most three are drawn (one per colour in `SERIES`); any further
182        models are ignored.
183    path : str or pathlib.Path
184        Output PNG.
185    bins : int, optional
186        Number of equal-width bins on [0, 1].
187    """
188    fig, ax = plt.subplots(figsize=(4.8, 4.4))
189    ax.plot([0, 1], [0, 1], color=MUTED, linewidth=1, linestyle="--", label="Perfect calibration")
190    edges = np.linspace(0, 1, bins + 1)
191    for (label, p), color in zip(curves.items(), SERIES):
192        # Same binning as `overtake.train.expected_calibration_error`; the clip puts p = 1 in the last bin.
193        idx = np.clip(np.digitize(p, edges) - 1, 0, bins - 1)
194        # Below 20 attempts a bin's observed rate is too noisy to plot (threshold: estimate).
195        keep = [b for b in range(bins) if (idx == b).sum() >= 20]
196        ax.plot([p[idx == b].mean() for b in keep], [y[idx == b].mean() for b in keep],
197                color=color, label=label, **MARKER)
198    _percent_axis(ax, "x")
199    _percent_axis(ax)
200    ax.set_xlabel("Predicted probability")
201    ax.set_ylabel("Observed overtake rate")
202    ax.set_title("Calibration on the test set")
203    ax.legend(loc="upper left")
204    _save(fig, path)

207def what_if(predict, path, exact=None, base=None):
208    """How the probability responds to gap, battery charge, straight length and Boost.
209
210    Left: probability against the gap (0.05-2 s) for attacker charges of 25,
211    50, 75 and 100 %, with Overtake Mode on only within 1 s as the rules
212    allow; the base scenario is marked. Right: probability against the length
213    of the straight over 250-1300 m (the range the training data covers for
214    straights), with and without the attacker's Boost. Every other input stays
215    at ``base``.
216
217    Parameters
218    ----------
219    predict : callable
220        Maps a list of scenario dicts to an array of probabilities (e.g.
221        ``OvertakePredictor.predict``).
222    path : str or pathlib.Path
223        Output PNG.
224    exact : callable, optional
225        Same signature, giving the simulator's true probability (e.g. a
226        wrapper around `overtake.simulator.probability`); drawn dashed for
227        comparison.
228    base : dict, optional
229        Fields that override `overtake.features.DEFAULTS` in the scenario
230        varied around.
231
232    Raises
233    ------
234    ValueError
235        If ``predict`` or ``exact`` rejects a scenario;
236        ``OvertakePredictor.predict`` does so for an unknown field or an
237        out-of-range value in ``base``.
238    """
239    base = {**DEFAULTS, **(base or {})}
240    fig, (left, right) = plt.subplots(1, 2, figsize=(10, 3.9), sharey=True)
241
242    def draw(ax, xs, scenarios, color, label):
243        """Plot ``predict`` over ``scenarios`` against ``xs`` as a solid line and, if given, ``exact`` dashed."""
244        ax.plot(xs, predict(scenarios), color=color, label=label)
245        if exact is not None:
246            ax.plot(xs, exact(scenarios), color=color, linewidth=1, linestyle="--")
247
248    gaps = np.linspace(0.05, 2.0, 80)
249    for battery, color in zip((25, 50, 75, 100), BATTERY_RAMP):
250        # Overtake Mode needs a gap of at most 1 s at the detection point, so it switches off beyond 1 s.
251        draw(left, gaps, [{**base, "gap_s": g, "overtake_mode": g <= 1.0, "attacker_battery": battery}
252                          for g in gaps], color, f"{battery}% battery")
253    p_base = float(np.asarray(predict([base]))[0])
254    left.plot([base["gap_s"]], [p_base], marker="o", markersize=8, color=BATTERY_RAMP[-1],
255              markeredgecolor=SURFACE, markeredgewidth=2, zorder=5)
256    left.annotate(f"Default scenario: {p_base:.0%}", (base["gap_s"], p_base), xytext=(10, 0),
257                  textcoords="offset points", va="center", color=INK)
258    left.axvline(1.0, color=AXIS, linewidth=1)
259    # Right of the line, between the legend and the default-scenario label, where no curve reaches.
260    left.annotate("Overtake Mode\nlimit (1 s)", (1.0, 0.535), xytext=(6, 0), textcoords="offset points",
261                  ha="left", va="center", color=MUTED, fontsize=11)
262    left.set_xlabel("Gap to the car ahead (s)")
263    left.set_ylabel("P(overtake)")
264    left.set_title("Gap and attacker battery charge")
265    left.legend(loc="upper right")
266    _percent_axis(left)  # also applies to the right panel (shared y)
267
268    # 250-1300 m: the straights the training data covers (`overtake.features.TRAINED_RUN_LENGTH_M`).
269    lengths = np.linspace(250, 1300, 80)
270    for boost, label, color in ((True, "With Boost", SERIES[0]), (False, "Without Boost", SERIES[1])):
271        draw(right, lengths, [{**base, "run_length_m": L, "attacker_boost": boost} for L in lengths], color, label)
272    right.set_xlabel("Length of the straight (m)")
273    right.set_title(f"Straight length, {base['gap_s']:g} s gap")
274    right.legend(loc="upper left")
275    if exact is not None:
276        right.annotate("Solid: neural network\nDashed: simulator truth", (0.02, 0.62),
277                       xycoords="axes fraction", ha="left", va="top", color=MUTED)
278    _save(fig, path)

281def convergence(results, path):
282    """Error against time step for Euler and RK4 on problems with analytical solutions.
283
284    One log-log panel per problem. Dashed grey lines of slope 1 and 4 through
285    each method's error at the last (smallest) step show the theoretical
286    orders.
287
288    Parameters
289    ----------
290    results : dict
291        `overtake.experiments.solver` output: ``{problem_title: {"dt": [...],
292        "euler": [...], "rk4": [...], "euler_order": float, "rk4_order":
293        float}}``, with steps in s listed from largest to smallest, final
294        position errors in m and the observed orders (least-squares slopes)
295        quoted in the legend.
296    path : str or pathlib.Path
297        Output PNG.
298    """
299    fig, axes = plt.subplots(1, len(results), figsize=(10, 3.8))
300    # atleast_1d: with one problem, plt.subplots returns a single Axes rather than an array.
301    for ax, (title, r) in zip(np.atleast_1d(axes), results.items()):
302        dt = np.array(r["dt"])
303        for method, label, color, order in (("rk4", "RK4", SERIES[0], 4), ("euler", "Euler", SERIES[1], 1)):
304            err = np.array(r[method])
305            ax.loglog(dt, err, color=color, label=f"{label} (observed order {r[method + '_order']:.2f})", **MARKER)
306            # Theoretical error C * dt^order, with C set so the line passes through the smallest step's error.
307            ref = err[-1] * (dt / dt[-1]) ** order
308            ax.loglog(dt, ref, color=MUTED, linewidth=1, linestyle="--")
309        _log_ticks(ax, dt)
310        ax.set_xlabel("Time step (s)")
311        ax.set_ylabel("Position error at end (m)")
312        ax.set_title(title)
313        ax.legend(loc="lower right")
314    np.atleast_1d(axes)[0].annotate("Dashed: slopes 1 and 4", (0.03, 0.95), xycoords="axes fraction",
315                                    va="top", color=MUTED)
316    _save(fig, path)

319def timestep(results, path, chosen_dt):
320    """Error and cost of the full two-car model against time step.
321
322    Left: the largest braking-point deficit error over the test scenarios
323    against time step, with the 1 ms acceptance limit. Right: the same error
324    against run time, i.e. accuracy against cost. The chosen step is
325    highlighted in both.
326
327    Parameters
328    ----------
329    results : dict
330        `overtake.experiments.timestep` output: ``{"dt": [...], "rk4": {...},
331        "euler": {...}}``, where each method has ``max_error_s`` (largest
332        deficit error against the fine-step reference, s) and ``seconds``
333        (run time for all the test scenarios, s); optionally
334        ``rk4_linear_crossing`` in the same form (RK4 with a linear instead
335        of a Hermite braking-point crossing; drawn dashed, accuracy only).
336    path : str or pathlib.Path
337        Output PNG.
338    chosen_dt : float
339        Step used by the simulator (`overtake.simulator.TIME_STEP_S`),
340        highlighted; must be one of ``results["dt"]``.
341
342    Raises
343    ------
344    ValueError
345        If ``chosen_dt`` is not in ``results["dt"]``.
346    """
347    fig, (left, right) = plt.subplots(1, 2, figsize=(10, 3.8))
348    dt = np.array(results["dt"])
349    for method, label, color in (("rk4", "RK4", SERIES[0]), ("euler", "Euler", SERIES[1])):
350        err, sec = np.array(results[method]["max_error_s"]), np.array(results[method]["seconds"])
351        left.loglog(dt, err, color=color, label=label, **MARKER)
352        right.loglog(sec, err, color=color, label=label, **MARKER)
353    if "rk4_linear_crossing" in results:
354        left.loglog(dt, results["rk4_linear_crossing"]["max_error_s"], color=SERIES[0], linewidth=1,
355                    linestyle="--", label="RK4, linear crossing")
356    i = list(dt).index(chosen_dt)  # exact match: the chosen step must be one of the studied steps
357    # Label below-right of the point on the left panel (the curves rise to the right there), left of it on the right.
358    for ax, x, offset, ha, va in ((left, chosen_dt, (8, -10), "left", "top"),
359                                  (right, results["rk4"]["seconds"][i], (-10, 0), "right", "center")):
360        ax.plot([x], [results["rk4"]["max_error_s"][i]], marker="o", markersize=10, color=SERIES[0],
361                markeredgecolor=SURFACE, markeredgewidth=2, zorder=5)
362        ax.annotate(f"chosen: dt = {chosen_dt} s", (x, results["rk4"]["max_error_s"][i]), xytext=offset,
363                    textcoords="offset points", ha=ha, va=va, color=INK)
364        ax.axhline(1e-3, color=AXIS, linewidth=1)  # 1 ms limit on the deficit error
365    left.annotate("1 ms acceptance limit", (dt.min(), 1e-3), xytext=(0, 4), textcoords="offset points",
366                  ha="left", color=MUTED)
367    _log_ticks(left, sorted(dt))
368    left.set_xlabel("Time step (s)")
369    left.set_ylabel("Max deficit error (s)")
370    left.set_title("Accuracy of the braking-point deficit")
371    left.legend(loc="upper left")
372    right.set_xlabel("Run time for the test scenarios (s)")
373    right.set_title("Cost against accuracy")
374    right.legend(loc="upper right")
375    _save(fig, path)

378def stability(results, path):
379    """Energy-budget overshoot against time step, Euler and RK4, with the predicted stability limits.
380
381    One panel per battery deployment. Near an empty battery the deployed
382    power ramps down as a fast decay with rate lambda, which an explicit
383    method follows stably only for dt below its real-axis stability bound
384    divided by lambda (2 / lambda for Euler, 2.785 / lambda for RK4). These
385    predicted limits are drawn dashed in each method's colour, the chosen step
386    in grey.
387
388    Parameters
389    ----------
390    results : dict
391        `overtake.experiments.stability` output: ``dt`` (steps, s),
392        ``chosen_dt`` (s) and ``deployments``, a dict of panel title ->
393        ``{"decay_rate_per_s": float, "predicted_dt_limit_s": {"euler":
394        float, "rk4": float}, "euler": {"max_energy_overshoot_j": [...]},
395        "rk4": {...}}``. Needs at least two deployments.
396    path : str or pathlib.Path
397        Output PNG.
398
399    Raises
400    ------
401    TypeError

404    """
405    deployments = results["deployments"]
406    fig, axes = plt.subplots(1, len(deployments), figsize=(10, 3.6), sharey=True)
407    dt = np.array(results["dt"])
408    for ax, (title, row) in zip(axes, deployments.items()):
409        for method, label, color in (("rk4", "RK4", SERIES[0]), ("euler", "Euler", SERIES[1])):
410            # J -> kJ

412            limit = row["predicted_dt_limit_s"][method]

416                        xytext=(3, -3), textcoords="offset points", va="top", color=color, fontsize=10)
417        ax.axvline(results["chosen_dt"], color=AXIS, linewidth=1)

419                    xycoords=ax.get_xaxis_transform(), xytext=(4, 0), textcoords="offset points",
420                    ha="left", color=MUTED, fontsize=10)
421        ax.set_title(f"{title}: decay rate {row['decay_rate_per_s']:.1f} 1/s")
422        ax.set_xlabel("Time step (s)")
423    # One legend, in the empty space between the chosen step and the limits.
424    axes[-1].legend(loc="upper left", bbox_to_anchor=(0.2, 0.97))
425    axes[0].set_ylabel("Energy used beyond the battery (kJ)")
426    _save(fig, path)

429def quadrature(results, path):
430    """Quadrature rule convergence and Monte Carlo error.
431
432    Left: the largest error in P over the test scenarios against the number
433    of integrand evaluations, per rule. Right: the RMS error of the Monte Carlo
434    estimate against the number of replays, beside the binomial theory
435    sqrt(p(1-p)/N).
436
437    Parameters
438    ----------
439    results : dict
440        `overtake.experiments.quadrature` output: ``{"rules": {rule:
441        {"evaluations": [...], "max_error": [...]}}, "monte_carlo": {"runs":
442        [...], "rms_error": [...], "theory": [...]}}``. At most three rules
443        are drawn (one per colour in `SERIES`).
444    path : str or pathlib.Path
445        Output PNG.
446    """
447    fig, (left, right) = plt.subplots(1, 2, figsize=(10, 3.8))
448    for (rule, r), color in zip(results["rules"].items(), SERIES):
449        err = np.maximum(r["max_error"], 1e-17)  # floor so an error of exactly 0 can still go on a log axis
450        left.loglog(r["evaluations"], err, color=color, label=rule, **MARKER)
