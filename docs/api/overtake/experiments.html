 55def _timed(func, repeats=3):
 56    """Best wall-clock time of ``repeats`` calls, and the last result."""
 57    best = float("inf")
 58    for _ in range(repeats):
 59        start = time.perf_counter()
 60        out = func()
 61        best = min(best, time.perf_counter() - start)
 62    return best, out

 71def nn_flops(hidden, n_inputs=None):
 72    """Floating-point operations for one network prediction.
 73
 74    Counts a multiply-add as 2 FLOPs, a bias add as 1 and a SiLU activation as
 75    about 5 (exponential, add, divide, multiply). Embedding look-ups are memory
 76    reads, not arithmetic.
 77
 78    Parameters
 79    ----------
 80    hidden : tuple of int
 81    n_inputs : int, optional
 82        Width of the first layer's input (embeddings + numeric inputs).
 83
 84    Returns
 85    -------
 86    int
 87    """
 88    if n_inputs is None:
 89        n_inputs = OvertakeNet(hidden).mlp[0].in_features
 90    widths = [n_inputs, *hidden, 1]
 91    linear = sum(2 * a * b + b for a, b in zip(widths[:-1], widths[1:]))
 92    return linear + 5 * sum(hidden)
 93
 94
 95# Hand count of the arithmetic in physics.two_car_rhs: about 35 FLOPs per car
 96# in `acceleration` plus about 20 for the slipstream (an exponential counted
 97# as ~10). RK4 adds 4 right-hand-side evaluations and ~14 vector operations
 98# per state variable per step.
 99FLOPS_PER_RHS = 2 * 35 + 20
100FLOPS_PER_RK4_STEP = 4 * FLOPS_PER_RHS + 6 * 14
101FLOPS_QUADRATURE = 16 * 30 + 40  # 16 Gauss nodes (density + logistic) and the closed-form tail

106def _single_car(car):
107    state = CarState(mu=np.array([1e9]), cda_m2=np.array([car.cda_m2]),
108                     deploy_share=np.array([0.0]), energy_j=np.array([0.0]))
109
110    def f(y):
111        accel, ers = acceleration(y[1:2], y[2:3], state, state.cda_m2, car)
112        return np.array([y[1], accel[0], ers[0]])
113    return f

116def solver():
117    """Verify Euler and RK4 on two special cases of the car model with exact solutions.
118
119    * Constant power ``P``, no resistance: ``v^2 = v0^2 + 2 P t / m`` and
120      ``x = m / (3P) (v^3 - v0^3)``.
121    * Drag only: ``v = v0 / (1 + k v0 t / m)`` and ``x = (m / k) ln(1 + k v0 t / m)``.
122
123    The observed order of convergence is the slope of log(error) against
124    log(dt) over the smallest steps.
125    """
126    m, p, k, v0, t_end = 800.0, 500e3, 0.6, 30.0, 12.8
127    power_only = replace(CarParameters(), mass_kg=m, ice_power_w=p, cda_m2=0.0, cla_m2=0.0,
128                         rolling_coeff=0.0, drivetrain_efficiency=1.0)
129    drag_only = replace(power_only, ice_power_w=0.0, cda_m2=2 * k / 1.2, air_density=1.2)
130    problems = {
131        "Constant power, no drag": (power_only, m / (3 * p) * ((v0**2 + 2 * p * t_end / m) ** 1.5 - v0**3)),
132        "Drag only, no power": (drag_only, m / k * np.log(1 + k * v0 * t_end / m)),
133    }
134    dts = [0.8, 0.4, 0.2, 0.1, 0.05, 0.025, 0.0125]
135    out = {}
136    for title, (car, x_exact) in problems.items():
137        r = {"dt": dts, "exact_position_m": x_exact}
138        for method in ("euler", "rk4"):
139            r[method] = [abs(integrate(_single_car(car), [0.0, v0, 0.0], dt, int(round(t_end / dt)), method)[0]
140                             - x_exact) for dt in dts]
141            tail = slice(-4, None) if method == "euler" else slice(1, 5)  # asymptotic, above round-off
142            r[method + "_order"] = float(np.polyfit(np.log(dts[tail]), np.log(r[method][tail]), 1)[0])
143        out[title] = r
144    plots.convergence(out, FIGURES / "solver_convergence.png")
145    return _write("solver", out)

150def timestep(n=2000, dts=(0.4, 0.2, 0.1, 0.05, 0.025, 0.0125), reference_dt=0.001):
151    """Deficit error of the full model against a fine-step RK4 reference.
152
153    The full model has kinks (traction limit, MGU-K taper, energy running
154    out), so the observed order is lower than the smooth-case order; this
155    study chooses the step that keeps the error well below the 1 ms
156    acceptance limit (0.5 % of the 0.2 s execution noise).
157    """
158    df = simulator.sample_scenarios(n, np.random.default_rng(7))
159    reference = simulator.closing_run(df, dt=reference_dt)[0].deficit_s
160    out = {"dt": list(dts), "reference_dt": reference_dt, "scenarios": n}
161    for method in ("rk4", "euler"):
162        rows = {"max_error_s": [], "mean_error_s": [], "seconds": []}
163        for dt in dts:
164            seconds, (run, _, _) = _timed(lambda: simulator.closing_run(df, dt=dt, method=method), repeats=2)
165            err = np.abs(run.deficit_s - reference)
166            rows["max_error_s"].append(float(err.max()))
167            rows["mean_error_s"].append(float(err.mean()))
168            rows["seconds"].append(seconds)
169        fine = slice(1, None)
170        rows["observed_order"] = float(np.polyfit(np.log(dts[fine]), np.log(rows["mean_error_s"][fine]), 1)[0])
171        out[method] = rows
172    plots.timestep(out, FIGURES / "timestep_study.png", simulator.TIME_STEP_S)
173    return _write("timestep", out)

178def quadrature(n=2000, mc_scenarios=200):
179    """Accuracy of the quadrature rules and of Monte Carlo replays.
180
181    Reference: 200-node Gauss-Legendre (agrees with SciPy's adaptive ``quad``
182    to 1e-10, see ``tests/test_simulator.py``).
183    """
184    df = simulator.sample_scenarios(n, np.random.default_rng(11))
185    params = simulator.outcome_parameters(df, *simulator.closing_run(df))
186    reference = simulator.overtake_integral(*params, rule="gauss", n=200)
187    rules = {
188        "Gauss-Legendre": ("gauss", [2, 4, 8, 12, 16, 24, 32], lambda k: k),
189        "Simpson": ("simpson", [2, 4, 8, 16, 32, 64, 128, 256], lambda k: k + 1),
190        "Trapezoid": ("trapezoid", [2, 4, 8, 16, 32, 64, 128, 256], lambda k: k + 1),
191    }
192    out = {"rules": {}}
193    for title, (rule, sizes, evaluations) in rules.items():
194        out["rules"][title] = {
195            "n": sizes,
196            "evaluations": [evaluations(k) for k in sizes],
197            "max_error": [float(np.abs(simulator.overtake_integral(*params, rule=rule, n=k) - reference).max())
198                          for k in sizes],
199        }
200
201    sub = df.iloc[:mc_scenarios]
202    exact = reference[:mc_scenarios]
203    runs = [100, 400, 1600, 6400, 25600]
204    out["monte_carlo"] = {
205        "runs": runs,
206        "rms_error": [float(np.sqrt(np.mean((simulator.monte_carlo_probability(sub, r, seed=3) - exact) ** 2)))
207                      for r in runs],
208        "theory": [float(np.sqrt(np.mean(exact * (1 - exact)) / r)) for r in runs],
209    }
210    seconds_gauss, _ = _timed(lambda: simulator.overtake_integral(*params))
211    seconds_mc, _ = _timed(lambda: simulator.draw_outcomes(
212        *[np.repeat(a[:, None], 1600, axis=1) for a in params], np.random.default_rng(0)), repeats=1)
213    out["seconds_per_scenario"] = {"gauss_16": seconds_gauss / n, "monte_carlo_1600": seconds_mc / n}
214    plots.quadrature(out, FIGURES / "quadrature_study.png")
215    return _write("quadrature", out)

223def _inference_seconds(predictor, df, repeats=3):
224    cat, num = predictor.preprocessor.transform(df)
225    with torch.no_grad():
226        seconds, _ = _timed(lambda: [m(cat, num) for m in predictor.models], repeats)
227    return seconds

230def selection():
231    """Choose the network size (single networks, raw inputs), then check how much data it needs.
232
233    Selection uses the validation set only. Both validation log loss (which
234    would be available with real data) and error against the true
235    probability (simulation only) are recorded, to check they agree.
236
237    Rule: the one-standard-error rule (Hastie, Tibshirani and Friedman, *The
238    Elements of Statistical Learning*, Section 7.10) - the smallest network
239    whose validation log loss is within one standard error of the best. Two
240    runs with different data put 64-64 and 128-128-64 in different orders by
241    less than 0.001, so a rule that only takes the minimum would pick on noise.
242    """
243    train_df, val_df, _ = split(dataset(), SEED)
244    y, p_true = val_df[LABEL].to_numpy(), val_df["p_true"].to_numpy()
245    default_truth = float(simulator.probability(pd.DataFrame([DEFAULTS]))[0])
246    architectures, trained = [], {}
247    for hidden in ARCHITECTURES:
248        start = time.perf_counter()
249        predictor, history = train_network(train_df, val_df, hidden=hidden, seed=SEED, log=lambda _: None)
250        fit_seconds = time.perf_counter() - start
251        trained[hidden] = predictor
252        p = predictor.predict(val_df)
253        m = metrics(y, p, p_true)
254        p = np.clip(p, 1e-7, 1 - 1e-7)
255        losses = -(y * np.log(p) + (1 - y) * np.log(1 - p))
256        architectures.append({
257            "label": "-".join(map(str, hidden)),
258            "hidden": list(hidden),
259            "parameters": sum(p.numel() for p in predictor.model.parameters()),
260            "flops_per_prediction": nn_flops(hidden),
261            "epochs": len(history),
262            "fit_seconds": fit_seconds,
263            "inference_us_per_prediction": 1e6 * _inference_seconds(predictor, val_df) / len(val_df),
264            "val_log_loss": m["log_loss"],
265            "val_log_loss_se": float(losses.std(ddof=1) / np.sqrt(len(losses))),
266            "val_mae_vs_true": m["mae_vs_true"],
267            "default_scenario_error": predictor.predict(DEFAULTS) - default_truth,
268        })
269        print(architectures[-1], flush=True)
270
271    lowest = min(architectures, key=lambda a: a["val_log_loss"])
272    threshold = lowest["val_log_loss"] + lowest["val_log_loss_se"]
273    best = min((a for a in architectures if a["val_log_loss"] <= threshold), key=lambda a: a["parameters"])
274    chosen = tuple(best["hidden"])
275
276    data_sizes = []
277    for size in (10_000, 30_000, 100_000, len(train_df)):
278        if size == len(train_df):
279            predictor = trained[chosen]
280        else:
281            predictor, _ = train_network(train_df.iloc[:size], val_df, hidden=chosen, seed=SEED, log=lambda _: None)
282        m = metrics(y, predictor.predict(val_df), p_true)
283        data_sizes.append({"train_size": size, "val_log_loss": m["log_loss"], "val_mae_vs_true": m["mae_vs_true"]})
284        print(data_sizes[-1], flush=True)
285
286    plots.model_selection(architectures, data_sizes, FIGURES / "model_selection.png")
287    return _write("selection", {"architectures": architectures, "lowest_log_loss": lowest["label"],
288                                "threshold": threshold, "selected": best["label"],
289                                "selection_rule": "smallest network within one standard error of the "
290                                                  "lowest validation log loss",
291                                "default_scenario_truth": default_truth, "data_sizes": data_sizes})

294def ablation(seeds=(0, 1, 2, 3, 4)):
295    """Physics-informed inputs and ensembling, over five training seeds.
296
297    The selected network is trained five times (seeds 0-4) with and without
298    the `DERIVED` energy-cover inputs. For each single run, and for the
299    ensemble that averages the five runs' probabilities, this records the
300    validation log loss and error against truth, the error at the default
301    scenario, and the worst error along the attacker-battery axis through the
302    default scenario (30-100 %).
303
304    A first version with three seeds showed that the seed alone moves the
305    default-scenario prediction by about 3 points, which is what motivated
306    testing an ensemble.
307    """
308    from .train import HIDDEN
309    train_df, val_df, _ = split(dataset(), SEED)
310    y, p_true = val_df[LABEL].to_numpy(), val_df["p_true"].to_numpy()
311    axis = pd.DataFrame([{**DEFAULTS, "attacker_battery": b} for b in np.arange(30.0, 101.0, 2.5)])
312    axis_truth = simulator.probability(axis)
313    default_truth = float(simulator.probability(pd.DataFrame([DEFAULTS]))[0])
314
315    def evaluate(p_val, p_default, p_axis):
316        m = metrics(y, p_val, p_true)
317        return {"val_log_loss": m["log_loss"], "val_mae_vs_true": m["mae_vs_true"],
318                "default_scenario_error": float(p_default - default_truth),
319                "battery_axis_max_error": float(np.abs(p_axis - axis_truth).max())}
320
321    out = {"hidden": list(HIDDEN), "seeds": list(seeds), "default_scenario_truth": default_truth}
322    for name, physics in (("raw inputs only", False), ("with energy-cover inputs", True)):
323        runs, preds = [], []
324        for seed in seeds:
325            predictor, history = train_network(train_df, val_df, hidden=HIDDEN, seed=seed, physics_features=physics,
326                                               log=lambda _: None)
327            p = (predictor.predict(val_df), predictor.predict(DEFAULTS), predictor.predict(axis))
328            preds.append(p)
329            runs.append({"seed": seed, "epochs": len(history), **evaluate(*p)})
330            print(name, runs[-1], flush=True)
331        summary = {key: {"mean": float(np.mean([r[key] for r in runs])), "sd": float(np.std([r[key] for r in runs], ddof=1))}
332                   for key in ("val_log_loss", "val_mae_vs_true", "default_scenario_error", "battery_axis_max_error")}
333        ensemble = evaluate(*(np.mean([p[i] for p in preds], axis=0) for i in range(3)))
334        out[name] = {"runs": runs, "single_model": summary, "ensemble_of_5": ensemble}
335        print(name, "ensemble", ensemble, flush=True)
336    return _write("ablation", out)

341def tyre_mismatch(df):
342    """True where either car's tyres do not suit the surface (slicks when damp/wet, treaded tyres when dry)."""
343    slicks = ["soft", "medium", "hard"]
344    dry = df["track_condition"] == "dry"
345    wrong = [(dry & ~df[c].isin(slicks)) | (~dry & df[c].isin(slicks)) for c in ("attacker_tyre", "defender_tyre")]
346    return (wrong[0] | wrong[1]).to_numpy()
347
348
349def _region_errors(test_df, p):
350    err = np.abs(p - test_df["p_true"].to_numpy())
351    regions = {}
352    for column in ("track_condition", "segment"):
353        for value in VOCABS[CATEGORICAL[column]]:
354            mask = (test_df[column] == value).to_numpy()
355            regions[value] = float(err[mask].mean())
356    mismatch = tyre_mismatch(test_df)
357    regions["tyres_suit_surface"] = float(err[~mismatch].mean())
358    regions["tyre_mismatch"] = float(err[mismatch].mean())
359    return regions
360
361
362def _error_quantiles(test_df, p):
363    err = np.abs(p - test_df["p_true"].to_numpy())
364    return {f"q{int(q * 1000) / 10:g}": float(np.quantile(err, q)) for q in (0.5, 0.9, 0.99, 0.999, 1.0)}

367def _model_bytes(obj):
368    if isinstance(obj, OvertakePredictor):
369        return sum(p.numel() * p.element_size() for m in obj.models for p in m.parameters())
370    return len(pickle.dumps(obj.pipeline))

373def comparison():
374    """Compare the network with the alternatives on the held-out test set.
375
376    Also tests extrapolation: straights of 1300-2000 m, longer than any in
377    training, where a data-driven model has no evidence.
378    """
379    train_df, _, test_df = split(dataset(), SEED)
380    y, p_true = test_df[LABEL].to_numpy(), test_df["p_true"].to_numpy()
381    network = OvertakePredictor.load()
382    models = {"Neural network": network,
383              "Logistic regression": baselines.logistic_regression().fit(train_df),
384              "Gradient boosting": baselines.gradient_boosting(SEED).fit(train_df)}
385    fit_seconds = {"Neural network": torch.load(DEFAULT_MODEL, map_location="cpu", weights_only=True)["fit_seconds"]}
386    fit_seconds.update({k: m.fit_seconds for k, m in models.items() if k != "Neural network"})
387
388    sample = test_df.iloc[:10_000]
389    out = {"test_attempts": len(test_df), "tyre_mismatch_share": float(tyre_mismatch(test_df).mean()),
390           "models": {}, "regions": {}, "error_quantiles": {}}
391    for name, model in models.items():
392        p = model.predict(test_df)
393        seconds, _ = _timed(lambda: model.predict(sample))
394        out["models"][name] = {**metrics(y, p, p_true), "fit_seconds": fit_seconds[name],
395                               "predict_us_per_scenario": 1e6 * seconds / len(sample),
396                               "model_bytes": _model_bytes(model)}
397        out["regions"][name] = _region_errors(test_df, p)
398        out["error_quantiles"][name] = _error_quantiles(test_df, p)
399        err, mismatch = np.abs(p - p_true), tyre_mismatch(test_df)
400        out["error_quantiles"][name]["tyre_mismatch_share_of_worst_1pct"] = float(
401            mismatch[err >= np.quantile(err, 0.99)].mean())
402        out["error_quantiles"][name]["tyre_mismatch_share_of_worst_0.1pct"] = float(
403            mismatch[err >= np.quantile(err, 0.999)].mean())
404
405    seconds, p_exact = _timed(lambda: simulator.probability(sample), repeats=1)
406    out["models"]["Direct simulation"] = {"mae_vs_true": float(np.abs(p_exact - sample["p_true"]).mean()),
407                                          "predict_us_per_scenario": 1e6 * seconds / len(sample),
408                                          "note": "the ground truth itself; needs the physical model, "
409                                                  "so cannot learn from observed race outcomes"}
410
411    rng = np.random.default_rng(21)
412    far = simulator.sample_scenarios(20_000, rng)
413    far = far[far["segment"] == "straight"].copy()
414    far["run_length_m"] = rng.uniform(1300, 2000, len(far))
415    truth = simulator.probability(far)
416    out["extrapolation_1300_2000m"] = {name: float(np.abs(m.predict(far) - truth).mean())
417                                       for name, m in models.items()}
418
419    plots.method_comparison(out["regions"], FIGURES / "method_comparison.png")
420    plots.calibration(y, {k: m.predict(test_df) for k, m in models.items()}, FIGURES / "calibration.png")
421    return _write("comparison", out)

426def performance():
427    """Where the time goes, and what vectorisation and chunking buy.
428
429    * cProfile of simulating 20,000 attempts.
430    * Vectorised integration of 200 scenarios together against one at a time.
431    * Chunk size for dataset generation (array size vs cache and overhead).
432    * FLOPs per prediction and achieved FLOP rate for the network and the
433      direct simulation; arithmetic intensity of network inference.
434    """
435    out = {}
436    profiler = cProfile.Profile()
437    profiler.enable()
438    simulator.generate_dataset(20_000, seed=1)
439    profiler.disable()
440    stats = pstats.Stats(profiler, stream=io.StringIO()).sort_stats("tottime")
441    top = sorted(stats.stats.items(), key=lambda kv: kv[1][2], reverse=True)[:8]
442    total = sum(v[2] for v in stats.stats.values())
443    out["profile_top_functions"] = [
444        {"function": f"{fn[0].split('/')[-1]}:{fn[2]}", "calls": v[1], "own_seconds": v[2],
445         "share": v[2] / total} for fn, v in top
446    ]
447
448    df = simulator.sample_scenarios(200, np.random.default_rng(4))
449    t_vec, _ = _timed(lambda: simulator.closing_run(df))

117def solver():
118    """Verify Euler and RK4 on two special cases of the car model with exact solutions.
119
120    * Constant power ``P``, no resistance: ``v^2 = v0^2 + 2 P t / m`` and
121      ``x = m / (3P) (v^3 - v0^3)``.
122    * Drag only: ``v = v0 / (1 + k v0 t / m)`` and ``x = (m / k) ln(1 + k v0 t / m)``.
123
124    The observed order of convergence is the slope of log(error) against
125    log(dt) over the smallest steps.
126    """
127    m, p, k, v0, t_end = 800.0, 500e3, 0.6, 30.0, 12.8
128    power_only = replace(CarParameters(), mass_kg=m, ice_power_w=p, cda_m2=0.0, cla_m2=0.0,
129                         rolling_coeff=0.0, drivetrain_efficiency=1.0)
130    drag_only = replace(power_only, ice_power_w=0.0, cda_m2=2 * k / 1.2, air_density=1.2)
131    problems = {
132        "Constant power, no drag": (power_only, m / (3 * p) * ((v0**2 + 2 * p * t_end / m) ** 1.5 - v0**3)),
133        "Drag only, no power": (drag_only, m / k * np.log(1 + k * v0 * t_end / m)),
134    }
135    dts = [0.8, 0.4, 0.2, 0.1, 0.05, 0.025, 0.0125]
136    out = {}
137    for title, (car, x_exact) in problems.items():
138        r = {"dt": dts, "exact_position_m": x_exact}
139        for method in ("euler", "rk4"):
140            r[method] = [abs(integrate(_single_car(car), [0.0, v0, 0.0], dt, int(round(t_end / dt)), method)[0]
141                             - x_exact) for dt in dts]
142            tail = slice(-4, None) if method == "euler" else slice(1, 5)  # asymptotic, above round-off
143            r[method + "_order"] = float(np.polyfit(np.log(dts[tail]), np.log(r[method][tail]), 1)[0])
144        out[title] = r
145    plots.convergence(out, FIGURES / "solver_convergence.png")
146    return _write("solver", out)

257        architectures.append({

343    """True where either car's tyres do not suit the surface (slicks when damp/wet, treaded tyres when dry)."""
344    slicks = ["soft", "medium", "hard"]
345    dry = df["track_condition"] == "dry"
346    wrong = [(dry & ~df[c].isin(slicks)) | (~dry & df[c].isin(slicks)) for c in ("attacker_tyre", "defender_tyre")]

515def sensitivity():
516    """Vary one input at a time between plausible extremes around the default scenario.
517
518    Compares the network's response with the simulator's, which checks that
519    the network learned the right sensitivities and not just the right
520    average. One-at-a-time analysis misses interactions; the what-if figure
521    shows two of the important ones.
522    """
523    network = OvertakePredictor.load()
524    rows = []
525    for column, (low, high, low_label, high_label) in SENSITIVITY_RANGES.items():
526        scenarios = [{**DEFAULTS, column: low}, {**DEFAULTS, column: high}]
527        if column == "gap_s":
528            for s in scenarios:
529                s["overtake_mode"] = s["gap_s"] <= 1.0
530        exact = simulator.probability(pd.DataFrame(scenarios))
531        rows.append({"input": column, "low": low_label, "high": high_label,
532                     "network": [float(v) for v in network.predict(scenarios)],
533                     "exact": [float(v) for v in exact]})
534    rows.sort(key=lambda r: abs(r["exact"][1] - r["exact"][0]), reverse=True)
535    plots.sensitivity(rows, FIGURES / "sensitivity.png")
536    plots.what_if(network.predict, FIGURES / "what_if.png",
537                  exact=lambda s: simulator.probability(pd.DataFrame(s)))
538    base = {"network": network.predict(DEFAULTS), "exact": float(simulator.probability(pd.DataFrame([DEFAULTS]))[0])}
539
540    # The bias near a full attacker battery, repeated on a wet track with intermediates.
541    wet = {**DEFAULTS, "track_condition": "wet", "attacker_tyre": "intermediate",
542           "defender_tyre": "intermediate", "track_temp_c": 20.0}
543    axis = pd.DataFrame([{**wet, "attacker_battery": b} for b in (40.0, 60.0, 80.0, 100.0)])
544    wet_axis = [{"attacker_battery": b, "network": float(n), "exact": float(e)}
545                for b, n, e in zip(axis["attacker_battery"], network.predict(axis), simulator.probability(axis))]
546    _, _, test_df = split(dataset(), SEED)
547    mask = ((test_df["track_condition"] == "wet") & (test_df["segment"] == "straight")
548            & test_df["attacker_tyre"].isin(["intermediate", "wet"]) & test_df["defender_tyre"].isin(["intermediate", "wet"]))
549    err = network.predict(test_df[mask]) - test_df.loc[mask, "p_true"].to_numpy()
550    wet_test = {"scenarios": int(mask.sum()), "mean_signed_error": float(err.mean()),
551                "mean_abs_error": float(np.abs(err).mean())}
552    return _write("sensitivity", {"default_scenario": base, "rows": rows,
553                                  "wet_intermediates_battery_axis": wet_axis,
554                                  "test_wet_straights_treaded_tyres": wet_test})
