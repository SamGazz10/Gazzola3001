  75def _timed(func, repeats=3):
  76    """Best wall-clock time (s) of ``repeats`` calls of ``func``, and the last call's result."""
  77    best = float("inf")
  78    for _ in range(repeats):
  79        start = time.perf_counter()
  80        out = func()
  81        # The fastest run is the one least disturbed by other work on the machine.
  82        best = min(best, time.perf_counter() - start)
  83    return best, out

 109@contextlib.contextmanager
 110def _patched(module, **values):
 111    """Temporarily replace module-level constants, restoring them afterwards.
 112
 113    Used only by the sensitivity studies of the model's constants and rules
 114    (`constants`, `rules`); the rest of the code never changes them.
 115
 116    Parameters
 117    ----------
 118    module : module
 119        For example `overtake.simulator`.
 120    **values
 121        Attribute name -> temporary value.
 122
 123    Yields
 124    ------
 125    None
 126    """
 127    old = {name: getattr(module, name) for name in values}
 128    try:
 129        for name, value in values.items():
 130            setattr(module, name, value)
 131        yield
 132    finally:
 133        for name, value in old.items():
 134            setattr(module, name, value)

 137def nn_flops(hidden, n_inputs=None):
 138    """Floating-point operations for one prediction by one network.
 139
 140    Counts a multiply-add as 2 FLOPs, a bias add as 1 and a SiLU activation as
 141    about 5 (exponential, add, divide, multiply). Embedding look-ups are memory
 142    reads, not arithmetic, and dropout is inactive at inference, so neither is
 143    counted.
 144
 145    Parameters
 146    ----------
 147    hidden : tuple of int
 148        Width of each hidden layer, e.g. ``(64, 64)``.
 149    n_inputs : int, optional
 150        Width of the first layer's input (embeddings + numeric inputs). By
 151        default that of an `overtake.model.OvertakeNet` on the raw inputs
 152        without the derived energy-cover inputs: 33 (19 embedding values,
 153        11 numbers and 3 flags).
 154
 155    Returns
 156    -------
 157    int
 158        FLOPs per prediction for a single network; multiply by the number of
 159        ensemble members for the whole predictor.
 160
 161    Examples
 162    --------
 163    >>> nn_flops((64, 64))
 164    13313
 165    """
 166    if n_inputs is None:
 167        n_inputs = OvertakeNet(hidden).mlp[0].in_features
 168    # Layer widths from input to the single output logit; consecutive pairs are the linear layers.
 169    widths = [n_inputs, *hidden, 1]
 170    linear = sum(2 * a * b + b for a, b in zip(widths[:-1], widths[1:]))
 171    return linear + 5 * sum(hidden)
 172

 174# Hand count (estimates) of the arithmetic per scenario, for the simulator's
 175# FLOP rate in `performance`. physics.two_car_rhs: about 35 FLOPs per car in
 176# `acceleration` plus about 20 for the slipstream (an exponential counted as
 177# ~10). An RK4 step makes 4 right-hand-side evaluations plus ~14 element-wise
 178# operations (stage sums and the weighted update) on each of the 6 state rows.
 179FLOPS_PER_RHS = 2 * 35 + 20
 180FLOPS_PER_RK4_STEP = 4 * FLOPS_PER_RHS + 6 * 14
 181# 16 Gauss nodes at ~30 FLOPs each (normal density + logistic) and ~40 for the closed-form tail (estimates).
 182FLOPS_QUADRATURE = 16 * 30 + 40

 187def _single_car(car):
 188    """Right-hand side ``f([x, v, e])`` of one car on its own, with grip and electrical power switched off."""
 189    # A huge friction coefficient removes the traction limit, and no deployment or energy removes the
 190    # MGU-K, so only the engine power and the resistances in ``car`` act.
 191    state = CarState(mu=np.array([1e9]), cda_m2=np.array([car.cda_m2]),
 192                     deploy_share=np.array([0.0]), energy_j=np.array([0.0]))
 193
 194    def f(y):
 195        """Time derivative of the single-car state [position m, speed m/s, energy used J]."""
 196        accel, ers = acceleration(y[1:2], y[2:3], state, state.cda_m2, car)
 197        return np.array([y[1], accel[0], ers[0]])
 198    return f

 201def solver():
 202    """Verify Euler and RK4 on two special cases of the car model with exact solutions.
 203
 204    * Constant power ``P``, no resistance (``m dv/dt = P / v``):
 205      ``v^2 = v0^2 + 2 P t / m`` and ``x = m / (3P) (v^3 - v0^3)``.
 206    * Drag only (``m dv/dt = -k v^2`` with ``k = rho CdA / 2``):
 207      ``v = v0 / (1 + k v0 t / m)`` and ``x = (m / k) ln(1 + k v0 t / m)``.
 208
 209    Both start at 30 m/s and run for 12.8 s, a whole number of every step
 210    tested (0.8 s halving down to 0.0125 s). The observed order of
 211    convergence is the slope of log(error) against log(dt) where the error
 212    is in its asymptotic regime and above round-off: the four smallest steps
 213    for Euler, and 0.4-0.05 s for RK4, whose drag-only errors at the two
 214    smallest steps are near double-precision round-off.
 215
 216    Returns
 217    -------
 218    dict
 219        Per problem: the steps, the exact final position (m), the absolute
 220        final-position error (m) of each method at each step, and each
 221        method's observed order.
 222    """
 223    m, p, k, v0, t_end = 800.0, 500e3, 0.6, 30.0, 12.8
 224    power_only = replace(CarParameters(), mass_kg=m, ice_power_w=p, cda_m2=0.0, cla_m2=0.0,
 225                         rolling_coeff=0.0, drivetrain_efficiency=1.0)
 226    # CdA chosen so that 1/2 rho CdA equals k with rho = 1.2 kg/m^3.
 227    drag_only = replace(power_only, ice_power_w=0.0, cda_m2=2 * k / 1.2, air_density=1.2)
 228    problems = {
 229        # (v0^2 + 2 P t / m)^1.5 is v^3 at t_end.
 230        "Constant power, no drag": (power_only, m / (3 * p) * ((v0**2 + 2 * p * t_end / m) ** 1.5 - v0**3)),
 231        "Drag only, no power": (drag_only, m / k * np.log(1 + k * v0 * t_end / m)),
 232    }
 233    dts = [0.8, 0.4, 0.2, 0.1, 0.05, 0.025, 0.0125]
 234    out = {}
 235    for title, (car, x_exact) in problems.items():
 236        r = {"dt": dts, "exact_position_m": x_exact}
 237        for method in ("euler", "rk4"):
 238            # Index 0 of the final state is the position; the state starts at [x = 0, v = v0, e = 0].
 239            r[method] = [abs(integrate(_single_car(car), [0.0, v0, 0.0], dt, int(round(t_end / dt)), method)[0]
 240                             - x_exact) for dt in dts]
 241            # Euler: the four smallest steps (at larger ones its constant-power error is not yet
 242            # asymptotic). RK4: 0.4-0.05 s,
 243            # leaving out the two smallest steps, whose drag-only errors are near round-off.
 244            tail = slice(-4, None) if method == "euler" else slice(1, 5)
 245            r[method + "_order"] = float(np.polyfit(np.log(dts[tail]), np.log(r[method][tail]), 1)[0])
 246        out[title] = r
 247    plots.convergence(out, FIGURES / "solver_convergence.png")
 248    return _write("solver", out)

 253def timestep(n=2000, dts=(0.4, 0.2, 0.1, 0.05, 0.025, 0.0125), reference_dt=0.001, fit_max_dt=0.1):
 254    """Deficit error of the full model against a fine-step RK4 reference.
 255
 256    The full model has kinks (traction limit, MGU-K taper, energy running
 257    out), so the observed order is lower than the smooth-case order; this
 258    study chooses the step that keeps the error well below the 1 ms
 259    acceptance limit (0.5 % of the 0.2 s execution noise).
 260
 261    Also measured:
 262
 263    * the observed order, fitted only over steps up to ``fit_max_dt``: larger
 264      steps (0.2 and 0.4 s) are beyond the stability limits on the battery
 265      ramp (0.114 s Euler, 0.159 s RK4; see `stability`), where the error is
 266      not in its asymptotic regime;
 267    * RK4 with straight-line instead of Hermite interpolation at the braking
 268      point, to measure what the fourth-order crossing buys;
 269    * the step and run time Euler would need, from a power-law fit of its
 270      worst error over the same steps, to meet the 1 ms limit and to match
 271      RK4's worst error at the chosen step.
 272
 273    Parameters
 274    ----------
 275    n : int
 276        Random scenarios (`overtake.simulator.sample_scenarios`, seed 7).
 277    dts : sequence of float
 278        Steps tested, s.
 279    reference_dt : float
 280        Step of the RK4 reference solution, s.
 281    fit_max_dt : float
 282        Largest step included in the order fits, s.
 283
 284    Returns
 285    -------
 286    dict
 287        Per method: worst and mean deficit error (s) and run time (s) per
 288        step, and the observed order of the mean and of the worst error; the
 289        same for RK4 with linear crossing; and, for each Euler target, the
 290        target error (s), the step needed (s) and its run time relative to
 291        RK4 at the chosen step.
 292
 293    Raises
 294    ------
 295    ValueError
 296        If ``dts`` does not contain the simulator's step,
 297        `overtake.simulator.TIME_STEP_S`.
 298    """
 299    df = simulator.sample_scenarios(n, np.random.default_rng(7))
 300    reference = simulator.closing_run(df, dt=reference_dt)[0].deficit_s
 301    # The small tolerance keeps a step equal to fit_max_dt despite floating-point representation.
 302    fit = [i for i, dt in enumerate(dts) if dt <= fit_max_dt + 1e-12]
 303    log_dt = np.log(np.asarray(dts)[fit])
 304    out = {"dt": list(dts), "reference_dt": reference_dt, "scenarios": n, "fit_dt": [dts[i] for i in fit]}
 305    for name, method, crossing in (("rk4", "rk4", "hermite"), ("euler", "euler", "hermite"),
 306                                   ("rk4_linear_crossing", "rk4", "linear")):
 307        rows = {"max_error_s": [], "mean_error_s": [], "seconds": []}
 308        for dt in dts:
 309            # closing_run returns (run, grip_a, grip_d); only the run is needed.
 310            seconds, (run, _, _) = _timed(
 311                lambda: simulator.closing_run(df, dt=dt, method=method, crossing=crossing), repeats=2)
 312            err = np.abs(run.deficit_s - reference)
 313            rows["max_error_s"].append(float(err.max()))
 314            rows["mean_error_s"].append(float(err.mean()))
 315            rows["seconds"].append(seconds)
 316        # Observed order: slope of log(error) against log(dt) over the fitted steps.
 317        for key, errors in (("observed_order", "mean_error_s"), ("observed_order_max_error", "max_error_s")):
 318            rows[key] = float(np.polyfit(log_dt, np.log(np.asarray(rows[errors])[fit]), 1)[0])
 319        out[name] = rows
 320
 321    # Euler's worst error as a power law of the step, err = exp(c) dt^p, fitted over the
 322    # same steps; its run time scales as 1/dt, so seconds x step is roughly constant.
 323    chosen = list(dts).index(simulator.TIME_STEP_S)
 324    p, c = np.polyfit(log_dt, np.log(np.asarray(out["euler"]["max_error_s"])[fit]), 1)
 325    finest = int(np.argmin(dts))
 326    # Seconds x step at the finest step, where the fixed per-run overheads matter least.
 327    euler_seconds_times_dt = out["euler"]["seconds"][finest] * dts[finest]
 328    rk4_seconds = out["rk4"]["seconds"][chosen]
 329    out["euler_equivalent"] = {}
 330    for key, target in (("for_1ms", 1e-3), ("to_match_rk4", out["rk4"]["max_error_s"][chosen])):
 331        # Invert the power law: dt = exp((log(err) - c) / p).
 332        dt_needed = float(np.exp((np.log(target) - c) / p))
 333        out["euler_equivalent"][key] = {"target_error_s": float(target), "dt_s": dt_needed,
 334                                        "cost_vs_rk4": float(euler_seconds_times_dt / dt_needed / rk4_seconds)}
 335    plots.timestep(out, FIGURES / "timestep_study.png", simulator.TIME_STEP_S)
 336    return _write("timestep", out)

 339def rk4_amplification(z):
 340    """RK4 amplification factor for the test equation ``y' = lambda y``.
 341
 342    One RK4 step multiplies the solution by ``R(z) = 1 + z + z^2/2 + z^3/6 + z^4/24``
 343    with ``z = lambda dt``; the step is stable while ``|R(z)| <= 1``.
 344
 345    Parameters
 346    ----------
 347    z : float or numpy.ndarray
 348
 349    Returns
 350    -------
 351    float or numpy.ndarray
 352    """
 353    return 1 + z + z**2 / 2 + z**3 / 6 + z**4 / 24

 356def stability(dts=(0.025, 0.05, 0.075, 0.1, 0.125, 0.15, 0.2, 0.25, 0.3), battery_j=0.2e6, t_end=3.0,
 357              reference_dt=0.001):
 358    """Stability of Euler and RK4 on the stiffest part of the model: the battery running out.
 359
 360    In the last ``energy_smoothing_j`` (20 kJ) of a battery the MGU-K power is
 361    ramped down in proportion to the energy left, so the energy used ``e``
 362    obeys ``de/dt = s P (E - e) / S`` (deployment share ``s``, MGU-K limit
 363    ``P``, energy available ``E``, ramp width ``S``). That is exponential
 364    decay with rate ``lambda = s P / S``: 17.5 1/s with Boost, against about
 365    0.25 1/s for the speed (drag and power). An explicit method is stable only
 366    while ``lambda dt`` lies inside its stability region, which on the real
 367    axis is 2 for Euler and about 2.785 for RK4, so this ramp, not the vehicle
 368    dynamics, sets the largest usable time step.
 369
 370    The check: one car starts at 200 km/h with only ``battery_j`` of energy
 371    (so the ramp is reached within a second) and is integrated for ``t_end``
 372    seconds at each step. An unstable step shows as the energy used
 373    overshooting the energy available, a physical constraint the exact
 374    solution never breaks. Positions are compared with a ``reference_dt``
 375    solution.
 376
 377    Parameters
 378    ----------
 379    dts : sequence of float
 380        Time steps to test, s.
 381    battery_j : float
 382        Energy available, J.
 383    t_end : float
 384        Integration time, s; must be a whole number of every step in ``dts``
 385        so that positions are compared at the same instant.
 386    reference_dt : float
 387        Step of the reference solution, s.
 388
 389    Returns
 390    -------
 391    dict
 392        Predicted step limits per deployment and method, and per step the
 393        largest energy overshoot (J) and final position error (m).
 394    """
 395    from scipy.optimize import brentq
 396
 397    steps = {dt: int(round(t_end / dt)) for dt in dts}
 398    if any(abs(n * dt - t_end) > 1e-9 for dt, n in steps.items()):
 399        raise ValueError("t_end must be a whole number of every time step")
 400    car = CarParameters()
 401    # Where |R(z)| = 1 on the negative real axis.
 402    rk4_limit = -brentq(lambda z: rk4_amplification(z) - 1, -3.0, -2.0)
 403    v0 = 200 / 3.6
 404    out = {"dt": list(dts), "battery_j": battery_j, "t_end_s": t_end, "chosen_dt": simulator.TIME_STEP_S,
 405           "real_axis_limit": {"euler": 2.0, "rk4": rk4_limit}, "deployments": {}}
 406    for title, share in (("Boost", 1.0), ("Standard map", simulator.STANDARD_MAP_SHARE)):
 407        state = CarState(mu=np.array([car.mu_reference]), cda_m2=np.array([car.cda_m2]),
 408                         deploy_share=np.array([share]), energy_j=np.array([battery_j]))
 409
 410        def f(y, state=state):
 411            """One car alone, state [x, v, e]; ``state`` is bound here so each deployment keeps its own."""
 412            accel, ers = acceleration(y[1:2], y[2:3], state, state.cda_m2, car)
 413            return np.array([y[1], accel[0], ers[0]])
 414
 415        rate = share * car.mguk_power_w / car.energy_smoothing_j
 416        reference = integrate(f, [0.0, v0, 0.0], reference_dt, int(round(t_end / reference_dt)))
 417        row = {"share": share, "decay_rate_per_s": rate,
 418               "predicted_dt_limit_s": {"euler": 2.0 / rate, "rk4": rk4_limit / rate}}
 419        for method, step in (("euler", euler_step), ("rk4", rk4_step)):
 420            overshoot, position_error = [], []
 421            for dt in dts:
 422                y, worst = np.array([0.0, v0, 0.0]), 0.0
 423                for _ in range(steps[dt]):
 424                    y = step(f, y, dt)
 425                    worst = max(worst, y[2] - battery_j)  # energy used beyond what the battery holds
 426                overshoot.append(float(worst))
 427                position_error.append(float(abs(y[0] - reference[0])))
 428            row[method] = {"max_energy_overshoot_j": overshoot, "final_position_error_m": position_error}
 429        out["deployments"][title] = row
 430    plots.stability(out, FIGURES / "stability.png")
 431    return _write("stability", out)

 434def adaptive(n=100, rtols=(1e-3, 1e-6, 1e-9), reference_dt=0.001, timing_n=20_000):
 435    """Fixed-step vectorised RK4 against SciPy's adaptive RK45 with event location.
 436
 437    An adaptive solver chooses its own steps to meet a tolerance and stops
 438    exactly at the braking point (an event), so it needs neither a time-step
 439    study nor the Hermite crossing. SciPy's ``solve_ivp`` integrates one
 440    scenario at a time, however, while the fixed-step RK4 integrates all of
 441    them together as arrays. Both are scored on the same scenarios against
 442    the 0.001 s RK4 reference used by `timestep`.
 443
 444    Parameters
 445    ----------
 446    n : int
 447        Scenarios for the accuracy comparison: the first ``n`` of the
 448        `timestep` sample (seed 7).
 449    rtols : sequence of float
 450        Relative tolerances given to ``solve_ivp``.
 451    reference_dt : float
 452        Step of the RK4 reference, s.
 453    timing_n : int
 454        Scenarios integrated together to time the fixed-step RK4 per scenario,
 455        as in dataset generation (with only ``n`` the fixed cost of each
 456        vectorised step would dominate).
 457
 458    Returns
 459    -------
 460    dict
 461        For the fixed-step RK4 and for each tolerance: worst deficit error
 462        (s), run time per scenario (microseconds) and right-hand-side
 463        evaluations per scenario.
 464    """
 465    from scipy.integrate import solve_ivp
 466
 467    df = simulator.sample_scenarios(2000, np.random.default_rng(7)).iloc[:n].reset_index(drop=True)
 468    reference = simulator.closing_run(df, dt=reference_dt)[0].deficit_s
 469    run = simulator.closing_run(df)[0]
 470    many = simulator.sample_scenarios(timing_n, np.random.default_rng(8))
 471    seconds, _ = _timed(lambda: simulator.closing_run(many))
 472    out = {"scenarios": n, "reference_dt": reference_dt, "timing_scenarios": timing_n,
 473           "rk4_fixed_step": {"dt": simulator.TIME_STEP_S,
 474                              "max_error_s": float(np.abs(run.deficit_s - reference).max()),
 475                              "us_per_scenario": 1e6 * seconds / timing_n,
 476                              # Four right-hand-side evaluations per step, plus two at the crossing.
 477                              "rhs_evaluations_per_scenario": float(4 * np.mean(run.time_s) / simulator.TIME_STEP_S + 2)},
 478           "solve_ivp_rk45": []}
 479
 480    ic = simulator.initial_conditions(df)
 481    # The states have different units and sizes (m, m/s, J); energy is about a
 482    # thousand times larger in magnitude, so its absolute tolerance is too.
 483    scale = np.array([1.0, 1.0, 1e3, 1.0, 1.0, 1e3])
 484
 485    def deficit(i, rtol):
 486        """Deficit (s) at the braking point of scenario ``i`` from ``solve_ivp``, and its RHS evaluations."""
 487        attacker, defender = ic["attacker"].take([i]), ic["defender"].take([i])
 488        length = ic["length"][i]
 489
 490        def rhs(t, y):
 491            """Two-car right-hand side for one scenario in the (t, y) form ``solve_ivp`` expects."""
 492            return two_car_rhs(y[:, None], attacker, defender, simulator.CAR)[:, 0]
 493
 494        def braking_point(t, y):
 495            """Event function: zero when the defender reaches the braking point."""
 496            return y[3] - length
 497        braking_point.terminal, braking_point.direction = True, 1
 498
 499        y0 = [-ic["gap"][i] * ic["v0_d"][i], ic["v0_a"][i], 0.0, 0.0, ic["v0_d"][i], 0.0]
 500        sol = solve_ivp(rhs, (0.0, 60.0), y0, method="RK45", rtol=rtol, atol=rtol * scale, events=braking_point)
 501        xa, va, _, xd, _, _ = sol.y_events[0][0]
 502        return (xd - xa) / max(va, 1.0), sol.nfev
 503
 504    for rtol in rtols:
 505        start = time.perf_counter()
 506        results = [deficit(i, rtol) for i in range(n)]
 507        elapsed = time.perf_counter() - start
 508        values = np.array([r[0] for r in results])
 509        out["solve_ivp_rk45"].append({"rtol": rtol, "max_error_s": float(np.abs(values - reference).max()),
 510                                      "us_per_scenario": 1e6 * elapsed / n,
 511                                      "rhs_evaluations_per_scenario": float(np.mean([r[1] for r in results]))})
 512    return _write("adaptive", out)

 517def quadrature(n=2000, mc_scenarios=200):
 518    """Accuracy of the quadrature rules and of Monte Carlo replays.
 519
 520    Reference: 200-node Gauss-Legendre (agrees with SciPy's adaptive ``quad``
 521    to 1e-10, see ``tests/test_simulator.py``).
 522    """
 523    df = simulator.sample_scenarios(n, np.random.default_rng(11))
 524    params = simulator.outcome_parameters(df, *simulator.closing_run(df))
 525    reference = simulator.overtake_integral(*params, rule="gauss", n=200)
 526    rules = {
 527        "Gauss-Legendre": ("gauss", [2, 4, 8, 12, 16, 24, 32], lambda k: k),
 528        "Simpson": ("simpson", [2, 4, 8, 16, 32, 64, 128, 256], lambda k: k + 1),
 529        "Trapezoid": ("trapezoid", [2, 4, 8, 16, 32, 64, 128, 256], lambda k: k + 1),
 530    }
 531    out = {"rules": {}}
 532    for title, (rule, sizes, evaluations) in rules.items():
 533        out["rules"][title] = {
 534            "n": sizes,
 535            "evaluations": [evaluations(k) for k in sizes],
 536            "max_error": [float(np.abs(simulator.overtake_integral(*params, rule=rule, n=k) - reference).max())
 537                          for k in sizes],
 538        }
 539
 540    sub = df.iloc[:mc_scenarios]
 541    exact = reference[:mc_scenarios]
 542    runs = [100, 400, 1600, 6400, 25600]
 543    out["monte_carlo"] = {
 544        "runs": runs,
 545        "rms_error": [float(np.sqrt(np.mean((simulator.monte_carlo_probability(sub, r, seed=3) - exact) ** 2)))
 546                      for r in runs],
 547        "theory": [float(np.sqrt(np.mean(exact * (1 - exact)) / r)) for r in runs],
 548    }
 549    seconds_gauss, _ = _timed(lambda: simulator.overtake_integral(*params))
 550    seconds_mc, _ = _timed(lambda: simulator.draw_outcomes(
 551        *[np.repeat(a[:, None], 1600, axis=1) for a in params], np.random.default_rng(0)), repeats=1)
 552    out["seconds_per_scenario"] = {"gauss_16": seconds_gauss / n, "monte_carlo_1600": seconds_mc / n}
 553    plots.quadrature(out, FIGURES / "quadrature_study.png")
 554    return _write("quadrature", out)

 623def _speed_dynamics(share, overtake_mode, car=simulator.CAR):
 624    """Where traction limits one car on new softs in the dry, and how fast its speed equation decays.
 625
 626    Parameters
 627    ----------
 628    share : float
 629        MGU-K deployment share (1 Boost, 0.55 standard map, 0 none).
 630    overtake_mode : bool
 631        Whether the Overtake Mode taper applies.
 632    car : CarParameters, optional
 633
 634    Returns
 635    -------
 636    dict
 637        ``traction_limited_below_kph``: speed where the power-limited and
 638        traction-limited drive forces are equal; ``max_speed_decay_rate_per_s``:
 639        the largest -d(acceleration)/dv between 150 and 350 km/h with energy
 640        available, the stiffness of the speed equation that `stability`
 641        compares with the battery ramp.
 642    """
 643    from scipy.optimize import brentq
 644
 645    state = CarState(mu=np.array([car.mu_reference]), cda_m2=np.array([car.cda_m2]),
 646                     deploy_share=np.array([share]), energy_j=np.array([1e9]),
 647                     overtake_mode=np.array([overtake_mode]))
 648
 649    def power_minus_traction(v):
 650        """Power-limited minus traction-limited drive force at speed ``v`` (m/s), N."""
 651        p_k = share * physics.mguk_power_limit(np.array([v]), np.array([overtake_mode]), car)[0]
 652        f_power = car.drivetrain_efficiency * (car.ice_power_w + p_k) / v
 653        f_traction = car.mu_reference * car.driven_axle_share * (
 654            car.mass_kg * physics.GRAVITY + 0.5 * car.air_density * car.cla_m2 * v**2)
 655        return f_power - f_traction
 656
 657    def accel(v):
 658        """Acceleration (m/s^2) at speed ``v`` with no energy used yet."""
 659        return acceleration(np.array([v]), np.array([0.0]), state, state.cda_m2, car)[0][0]
 660
 661    speeds = np.linspace(150, 350, 401) / 3.6
 662    h = 1e-3  # m/s; central difference of the acceleration
 663    rates = [-(accel(v + h) - accel(v - h)) / (2 * h) for v in speeds]
 664    return {"traction_limited_below_kph": 3.6 * brentq(power_minus_traction, 10.0, 90.0),
 665            "max_speed_decay_rate_per_s": float(max(rates))}

 686def constants(change=0.1, n=2000):
 687    """How much the exact probability depends on each estimated constant.
 688
 689    Each constant that is an estimate rather than a rule (`ESTIMATED_CONSTANTS`)
 690    is moved down and then up by ``change`` (10 %), the others staying at
 691    their values, and the exact probability is recomputed for the default
 692    scenario and for ``n`` random scenarios. This answers which assumptions
 693    matter most, which the input sensitivity (`sensitivity`) cannot.
 694
 695    Parameters
 696    ----------
 697    change : float
 698        Relative change applied to each constant.
 699    n : int
 700        Random scenarios (`overtake.simulator.sample_scenarios`, seed 31).
 701
 702    Returns
 703    -------
 704    dict
 705        Per constant: its value, the default-scenario probability at the low
 706        and high value, and the mean absolute change in probability over the
 707        random scenarios at each; sorted by the larger mean change.
 708    """
 709    car = simulator.CAR
 710    scenarios = pd.concat([pd.DataFrame([DEFAULTS]), simulator.sample_scenarios(n, np.random.default_rng(31))],
 711                          ignore_index=True)
 712    base = simulator.probability(scenarios)
 713
 714    def scaled(value, factor):
 715        """``value`` times ``factor``; for a table (dict) every entry is scaled."""
 716        # EXECUTION_SD is a table by track condition; scale every entry.
 717        return {k: v * factor for k, v in value.items()} if isinstance(value, dict) else value * factor
 718
 719    rows = []
 720    for label, where, name in ESTIMATED_CONSTANTS:
 721        value = getattr(car, name) if where == "car" else getattr(simulator, name)
 722        probabilities = []
 723        for factor in (1 - change, 1 + change):
 724            if where == "car":
 725                probabilities.append(simulator.probability(scenarios, car=replace(car, **{name: value * factor})))
 726            else:
 727                with _patched(simulator, **{name: scaled(value, factor)}):
 728                    probabilities.append(simulator.probability(scenarios))
 729        rows.append({"constant": label, "name": name, "value": value,
 730                     "default_scenario": [float(q[0]) for q in probabilities],
 731                     "mean_abs_change": [float(np.abs(q[1:] - base[1:]).mean()) for q in probabilities]})
 732    rows.sort(key=lambda r: max(r["mean_abs_change"]), reverse=True)
 733    out = {"change": change, "scenarios": n, "default_scenario": float(base[0]), "rows": rows}
 734    plots.constants(out, FIGURES / "constants.png")
 735    return _write("constants", out)

 739          batteries=(30, 60, 100)):
 740    """How the exact probability responds to the two Overtake Mode parameters.
 741
 742    Overtake Mode gives the attacker extra energy (0.5 MJ) and moves the start
 743    of the MGU-K taper from 290 to 337 km/h. Each is varied with the other at
 744    its 2026 value, for the default scenario at several attacker charge

 747    Parameters
 748    ----------
 749    energies_mj : sequence of float

 766        2026 values of both parameters, and the ``fia_issue17_taper``

 796    plots.rules(out, FIGURES / "rules.png")

 800def fia_issue17_power_limit(speed, overtake_mode, car=CarParameters()):
 801    """MGU-K power limit of the FIA 2026 F1 Technical Regulations, Issue 17, Article C5.2.8.
 802
 803    Without Overtake Mode: 1800 - 5 v kW below 340 km/h (350 kW up to 290 km/h,
 804    100 kW at 340), then 6900 - 20 v kW to zero at 345 km/h. With Overtake
 805    Mode: 7100 - 20 v kW (350 kW up to 337.5 km/h) to zero at 355 km/h. Both
 806    are capped at ``car.mguk_power_w`` (350 kW, Article C5.2.7). Used by
 807    `rules` to measure how much the model's simpler linear taper matters.
 808
 809    Parameters
 810    ----------
 811    speed : array_like
 812        Car speed, m/s.
 813    overtake_mode : array_like of bool
 814        Whether Overtake Mode is active.
 815    car : CarParameters, optional
 816
 817    Returns
 818    -------
 819    numpy.ndarray
 820        Power limit, W.
 821    """
 822    kph = np.asarray(speed) * 3.6
 823    standard = np.where(kph < 340, 1800 - 5 * kph, np.where(kph < 345, 6900 - 20 * kph, 0.0))
 824    overtake = np.where(kph < 355, 7100 - 20 * kph, 0.0)
 825    limit_kw = np.where(np.broadcast_to(overtake_mode, kph.shape), overtake, standard)
 826    return np.clip(limit_kw * 1e3, 0.0, car.mguk_power_w)

 834def _inference_seconds(predictor, df, repeats=3):
 835    """Best wall-clock seconds for every ensemble member's forward pass on ``df`` (preprocessing excluded)."""
 836    cat, num = predictor.preprocessor.transform(df)
 837    with torch.no_grad():
 838        seconds, _ = _timed(lambda: [m(cat, num) for m in predictor.models], repeats)
 839    return seconds

 842def selection():
 843    """Choose the network size (single networks, raw inputs), then check how much data it needs.
 844
 845    Selection uses the validation set only. Both validation log loss (which
 846    would be available with real data) and error against the true
 847    probability (simulation only) are recorded, to check they agree.
 848
 849    Rule: the one-standard-error rule (Hastie, Tibshirani and Friedman, *The
 850    Elements of Statistical Learning*, Section 7.10) - the smallest network
 851    whose validation log loss is within one standard error of the best. Two
 852    runs with different data put 64-64 and 128-128-64 in different orders by
 853    less than 0.001, so a rule that only takes the minimum would pick on noise.
 854    """
 855    train_df, val_df, _ = split(dataset(), SEED)
 856    y, p_true = val_df[LABEL].to_numpy(), val_df["p_true"].to_numpy()
 857    default_truth = float(simulator.probability(pd.DataFrame([DEFAULTS]))[0])
 858    architectures, trained = [], {}
 859    for hidden in ARCHITECTURES:
 860        start = time.perf_counter()
 861        predictor, history = train_network(train_df, val_df, hidden=hidden, seed=SEED, log=lambda _: None)
 862        fit_seconds = time.perf_counter() - start
 863        trained[hidden] = predictor
 864        p = predictor.predict(val_df)
 865        m = metrics(y, p, p_true)
 866        p = np.clip(p, 1e-7, 1 - 1e-7)
 867        losses = -(y * np.log(p) + (1 - y) * np.log(1 - p))
 868        architectures.append({
 869            "label": "-".join(map(str, hidden)),
 870            "hidden": list(hidden),
 871            "parameters": sum(p.numel() for p in predictor.model.parameters()),
 872            "flops_per_prediction": nn_flops(hidden),
 873            "epochs": len(history),
 874            "fit_seconds": fit_seconds,
 875            "inference_us_per_prediction": 1e6 * _inference_seconds(predictor, val_df) / len(val_df),
 876            "val_log_loss": m["log_loss"],
 877            "val_log_loss_se": float(losses.std(ddof=1) / np.sqrt(len(losses))),
 878            "val_mae_vs_true": m["mae_vs_true"],
 879            "default_scenario_error": predictor.predict(DEFAULTS) - default_truth,
 880        })
 881        print(architectures[-1], flush=True)
 882
 883    lowest = min(architectures, key=lambda a: a["val_log_loss"])
 884    threshold = lowest["val_log_loss"] + lowest["val_log_loss_se"]
 885    best = min((a for a in architectures if a["val_log_loss"] <= threshold), key=lambda a: a["parameters"])
 886    chosen = tuple(best["hidden"])
 887
 888    data_sizes = []
 889    for size in (10_000, 30_000, 100_000, len(train_df)):
 890        if size == len(train_df):
 891            predictor = trained[chosen]
 892        else:
 893            predictor, _ = train_network(train_df.iloc[:size], val_df, hidden=chosen, seed=SEED, log=lambda _: None)
 894        m = metrics(y, predictor.predict(val_df), p_true)
 895        data_sizes.append({"train_size": size, "val_log_loss": m["log_loss"], "val_mae_vs_true": m["mae_vs_true"]})
 896        print(data_sizes[-1], flush=True)
 897
 898    plots.model_selection(architectures, data_sizes, FIGURES / "model_selection.png")

 906def ablation(seeds=(0, 1, 2, 3, 4)):
 907    """Physics-informed inputs and ensembling, over five training seeds.
 908
 909    The selected network is trained five times (seeds 0-4) with and without
 910    the `DERIVED` energy-cover inputs. For each single run, and for the
 911    ensemble that averages the five runs' probabilities, this records the
 912    validation log loss and error against truth, the error at the default
 913    scenario, and the worst error along the attacker-battery axis through the
 914    default scenario (30-100 %).
 915
 916    A first version with three seeds showed that the seed alone moves the
 917    default-scenario prediction by about 3 points, which is what motivated
 918    testing an ensemble.
 919    """
 920    from .train import HIDDEN
 921    train_df, val_df, _ = split(dataset(), SEED)
 922    y, p_true = val_df[LABEL].to_numpy(), val_df["p_true"].to_numpy()
 923    axis = pd.DataFrame([{**DEFAULTS, "attacker_battery": b} for b in np.arange(30.0, 101.0, 2.5)])
 924    axis_truth = simulator.probability(axis)
 925    default_truth = float(simulator.probability(pd.DataFrame([DEFAULTS]))[0])
 926
 927    def evaluate(p_val, p_default, p_axis):
 928        """Validation metrics, default-scenario error and worst battery-axis error for one set of predictions."""
 929        m = metrics(y, p_val, p_true)
 930        return {"val_log_loss": m["log_loss"], "val_mae_vs_true": m["mae_vs_true"],
 931                "default_scenario_error": float(p_default - default_truth),
 932                "battery_axis_max_error": float(np.abs(p_axis - axis_truth).max())}
 933
 934    out = {"hidden": list(HIDDEN), "seeds": list(seeds), "default_scenario_truth": default_truth}
 935    for name, energy_cover in (("raw inputs only", False), ("with energy-cover inputs", True)):
 936        runs, preds = [], []
 937        for seed in seeds:
 938            predictor, history = train_network(train_df, val_df, hidden=HIDDEN, seed=seed, physics_features=energy_cover,
 939                                               log=lambda _: None)
 940            p = (predictor.predict(val_df), predictor.predict(DEFAULTS), predictor.predict(axis))
 941            preds.append(p)
 942            runs.append({"seed": seed, "epochs": len(history), **evaluate(*p)})
 943            print(name, runs[-1], flush=True)
 944        summary = {key: {"mean": float(np.mean([r[key] for r in runs])), "sd": float(np.std([r[key] for r in runs], ddof=1))}
 945                   for key in ("val_log_loss", "val_mae_vs_true", "default_scenario_error", "battery_axis_max_error")}
 946        ensemble = evaluate(*(np.mean([p[i] for p in preds], axis=0) for i in range(3)))
 947        out[name] = {"runs": runs, "single_model": summary, "ensemble_of_5": ensemble}
 948        print(name, "ensemble", ensemble, flush=True)
 949    return _write("ablation", out)

 954def tune_baselines(learning_rates=(0.05, 0.1, 0.2), leaves=(15, 31, 63), cs=(0.1, 1.0, 10.0)):
 955    """Check that the baselines were not handicapped by their settings.
 956
 957    Gradient boosting is refitted over a grid of learning rate and tree size
 958    (up to 1,000 trees, early stopping on an internal 10 % split) and
 959    logistic regression over its regularisation strength ``C``, all on the
 960    training set and scored on the validation set. The test set is not used.
 961    The settings used in `comparison` and the network's validation scores are
 962    recorded alongside.
 963
 964    Parameters
 965    ----------
 966    learning_rates, leaves : sequence
 967        Gradient-boosting grid: learning rate and maximum leaves per tree.
 968    cs : sequence of float
 969        Inverse L2 regularisation strengths for logistic regression.
 970
 971    Returns
 972    -------
 973    dict
 974        One row per setting with validation log loss and error against the
 975        exact probability.
 976    """
 977    train_df, val_df, _ = split(dataset(), SEED)
 978    y, p_true = val_df[LABEL].to_numpy(), val_df["p_true"].to_numpy()
 979
 980    def score(model):
 981        """Fit time and validation log loss and error against the exact probability of a fitted baseline."""
 982        m = metrics(y, model.predict(val_df), p_true)
 983        return {"fit_seconds": model.fit_seconds, "val_log_loss": m["log_loss"], "val_mae_vs_true": m["mae_vs_true"]}
 984
 985    out = {"gradient_boosting": [], "logistic_regression": []}
 986    for rate in learning_rates:
 987        for leaf in leaves:
 988            model = baselines.gradient_boosting(SEED, learning_rate=rate, max_leaf_nodes=leaf, max_iter=1000)
 989            model.fit(train_df)
 990            out["gradient_boosting"].append({"learning_rate": rate, "max_leaf_nodes": leaf, "max_iter": 1000,
 991                                             "trees": int(model.pipeline[-1].n_iter_), **score(model)})
 992            print(out["gradient_boosting"][-1], flush=True)
 993    for c in cs:
 994        model = baselines.logistic_regression(C=c).fit(train_df)
 995        out["logistic_regression"].append({"C": c, **score(model)})
 996    used = baselines.gradient_boosting(SEED).fit(train_df)
 997    out["used"] = {"gradient_boosting": {"learning_rate": 0.1, "max_leaf_nodes": 31, "max_iter": 500,
 998                                         "trees": int(used.pipeline[-1].n_iter_), **score(used)},
 999                   "logistic_regression": {"C": 1.0, **score(baselines.logistic_regression().fit(train_df))}}
1000    network = OvertakePredictor.load()
1001    m = metrics(y, network.predict(val_df), p_true)
1002    out["network_validation"] = {"val_log_loss": m["log_loss"], "val_mae_vs_true": m["mae_vs_true"]}
1003    return _write("baselines", out)

1006def _bootstrap_interval(values, n_resamples=1000, seed=0):
1007    """95 % percentile bootstrap interval for the mean of ``values``."""
1008    rng = np.random.default_rng(seed)
1009    means = [values[rng.integers(0, len(values), len(values))].mean() for _ in range(n_resamples)]
1010    return [float(np.quantile(means, 0.025)), float(np.quantile(means, 0.975))]

1020def tyre_mismatch(df):
1021    """True where either car's tyres do not suit the surface (slicks when damp/wet, treaded tyres when dry)."""
1022    slicks = ["soft", "medium", "hard"]
1023    dry = df["track_condition"] == "dry"
1024    wrong = [(dry & ~df[c].isin(slicks)) | (~dry & df[c].isin(slicks)) for c in ("attacker_tyre", "defender_tyre")]
1025    return (wrong[0] | wrong[1]).to_numpy()

1028def _region_errors(test_df, p):
1029    """Mean absolute error against ``p_true`` per weather, per segment and with or without suited tyres."""
1030    err = np.abs(p - test_df["p_true"].to_numpy())
1031    regions = {}

1033        for value in VOCABS[CATEGORICAL[column]]:

1055def comparison():
1056    """Compare the network with the alternatives on the held-out test set.

1061    model could reach on these 0/1 outcomes; the error in *changes* of
1062    probability for the what-if comparisons a strategist makes
1063    (`WHAT_IF_CHANGES`); and extrapolation to straights of 1300-2000 m,
1064    longer than any in training, where a data-driven model has no evidence.

1070              "Logistic regression": baselines.logistic_regression().fit(train_df),

1095    out["models"]["Direct simulation"] = {"mae_vs_true": float(np.abs(p_exact - sample["p_true"]).mean()),

1120                                       for name, m in models.items()}
1121
1122    plots.method_comparison(out["regions"], FIGURES / "method_comparison.png")
1123    plots.calibration(y, {k: m.predict(test_df) for k, m in models.items()}, FIGURES / "calibration.png")

1140      time of their own, and the cumulative time of each stage.

1142    * Chunk size for dataset generation (array size vs cache and overhead),
1143      and the peak memory of integrating one 20,000-scenario chunk

1145    * FLOPs per prediction and achieved FLOP rate for the network (batches of
1146      1,024 and 10,000) and the direct simulation; arithmetic intensity of

1174         "share": v[2] / total} for fn, v in top

1177    # the whole run goes (they overlap: closing_run contains the integration).

254def timestep(n=2000, dts=(0.4, 0.2, 0.1, 0.05, 0.025, 0.0125), reference_dt=0.001, fit_max_dt=0.1):
255    """Deficit error of the full model against a fine-step RK4 reference.
256
257    The full model has kinks (traction limit, MGU-K taper, energy running
258    out), so the observed order is lower than the smooth-case order; this
259    study chooses the step that keeps the error well below the 1 ms
260    acceptance limit (0.5 % of the 0.2 s execution noise).
261

263
264    * the observed order, fitted only over steps up to ``fit_max_dt``: larger

275    ----------
276    n : int

318        for key, errors in (("observed_order", "mean_error_s"), ("observed_order_max_error", "max_error_s")):

321
