overtake.predict

Predict the probability of an overtake for a race situation.

python -m overtake.predict --condition dry --tyre soft --segment straight --battery 100 --gap 1.0

From Python:

from overtake.predict import OvertakePredictor
predictor = OvertakePredictor.load()
predictor.predict({"track_condition": "dry", "attacker_tyre": "soft", "gap_s": 1.0})

Fields you leave out (or set to None) take the values in overtake.features.DEFAULTS, except the run length and track temperature, which depend on the segment and weather (overtake.features.situational_defaults), and overtake_mode, which defaults to "gap is 1 s or less". Invalid inputs raise a ValueError (the command line prints the message and exits with status 2; a missing model file exits with status 1).

  2
  3    python -m overtake.predict --condition dry --tyre soft --segment straight --battery 100 --gap 1.0
  4
  5From Python:
  6
  7    from overtake.predict import OvertakePredictor
  8    predictor = OvertakePredictor.load()
  9    predictor.predict({"track_condition": "dry", "attacker_tyre": "soft", "gap_s": 1.0})

 12`overtake.features.DEFAULTS`, except the run length and track temperature,

 14and ``overtake_mode``, which defaults to "gap is 1 s or less". Invalid inputs

 35class OvertakePredictor:
 36    """One or more trained networks together with the preprocessing they were trained with.
 37
 38    With several networks (an ensemble trained from different random seeds)
 39    the prediction is the mean of their probabilities.
 40
 41    Parameters
 42    ----------
 43    models : overtake.model.OvertakeNet or list of them
 44        All trained on the same data, so they share one preprocessor.
 45    preprocessor : overtake.features.Preprocessor
 46        Fitted on the networks' training data.
 47
 48    Attributes
 49    ----------
 50    models : list of overtake.model.OvertakeNet
 51        The networks, in evaluation mode (dropout off).
 52    preprocessor : overtake.features.Preprocessor
 53        The shared preprocessing.
 54    """
 55
 56    def __init__(self, models, preprocessor):
 57        """Store the networks in evaluation mode and the shared preprocessor (parameters as for the class)."""
 58        # Accept one network or an ensemble; eval() turns dropout off so predictions are deterministic.
 59        self.models = [m.eval() for m in (models if isinstance(models, (list, tuple)) else [models])]
 60        self.preprocessor = preprocessor
 61
 62    @property
 63    def model(self):
 64        """The first network (all members share the same architecture)."""
 65        return self.models[0]
 66
 67    @property
 68    def n_parameters(self):
 69        """Trainable parameters (embeddings, weights and biases) summed over all networks."""
 70        return sum(p.numel() for m in self.models for p in m.parameters())
 71
 72    @classmethod
 73    def load(cls, path=DEFAULT_MODEL):
 74        """Load a checkpoint written by `save`.
 75
 76        Parameters
 77        ----------
 78        path : str or pathlib.Path, optional
 79            Defaults to ``models/overtake_net.pt``.
 80
 81        Returns
 82        -------
 83        OvertakePredictor
 84
 85        Raises
 86        ------
 87        FileNotFoundError
 88            If there is no checkpoint at ``path``.
 89        KeyError
 90            If the file is not a checkpoint written by `save`.
 91
 92        Examples
 93        --------
 94        >>> predictor = OvertakePredictor.load()  # doctest: +SKIP
 95        >>> len(predictor.models), predictor.n_parameters  # doctest: +SKIP
 96        (5, 32945)
 97        """
 98        path = Path(path)
 99        if not path.exists():
100            raise FileNotFoundError(f"no trained model at {path}; run `python -m overtake.train` first")
101        # map_location="cpu": loads on machines without a GPU. weights_only=True: unpickles tensors and
102        # plain containers only, never arbitrary code, which is why `save` requires plain extras.
103        checkpoint = torch.load(path, map_location="cpu", weights_only=True)
104        models = []
105        for state in checkpoint["model_states"]:
106            model = OvertakeNet(checkpoint["hidden"], checkpoint["dropout"], checkpoint["n_inputs"])
107            model.load_state_dict(state)
108            models.append(model)
109        return cls(models, Preprocessor.from_state_dict(checkpoint["preprocessor"]))
110
111    def save(self, path, **extra):
112        """Write weights, architecture and preprocessing statistics to ``path``.
113
114        Parameters
115        ----------
116        path : str or pathlib.Path
117            Checkpoint file; missing parent folders are created.
118        **extra
119            Stored under their own keys (`overtake.train.main` adds
120            ``metrics``, ``history`` and ``fit_seconds``). Must be plain
121            Python values (such as dict, list, tuple, str, int, float, bool,
122            None), because `load` reads with ``torch.load(weights_only=True)``,
123            which refuses arbitrary pickled objects.
124
125        Notes
126        -----
127        Checkpoint keys: ``model_states`` (one state dict per network),
128        ``hidden``, ``dropout``, ``n_inputs`` and ``preprocessor``
129        (`overtake.features.Preprocessor.state_dict`), plus the ``extra`` keys.
130        """
131        Path(path).parent.mkdir(parents=True, exist_ok=True)
132        torch.save({
133            "model_states": [m.state_dict() for m in self.models],
134            "hidden": self.model.hidden,
135            "dropout": self.model.dropout,
136            "n_inputs": self.model.n_inputs,
137            "preprocessor": self.preprocessor.state_dict(),
138            **extra,
139        }, path)
140
141    def predict(self, scenarios):
142        """Probability of an overtake.
143
144        Parameters
145        ----------
146        scenarios : dict, pandas.Series, list of dict or pandas.DataFrame
147            One or more scenarios in the `overtake.features` schema. Missing
148            fields take their default values (see `complete`).
149
150        Returns
151        -------
152        float or numpy.ndarray of float32, shape (n,)
153            A float for a single scenario (dict or Series), otherwise one
154            probability per row. Either way the value is the mean of the
155            ensemble members' probabilities.
156
157        Raises
158        ------
159        ValueError
160            If a field name is unknown or a value is outside the schema (see
161            `overtake.features.validate`).
162
163        Examples
164        --------
165        With the committed model (the values change if it is retrained):
166
167        >>> predictor = OvertakePredictor.load()  # doctest: +SKIP
168        >>> round(predictor.predict({"gap_s": 0.5, "attacker_battery": 80}), 2)  # doctest: +SKIP
169        0.8
170        >>> predictor.predict([{"gap_s": 0.5}, {"gap_s": 1.5}]).round(2)  # doctest: +SKIP
171        array([0.78, 0.04], dtype=float32)
172        """
173        df = complete(scenarios)
174        cat, num = self.preprocessor.transform(df)
175        # Ensemble: the mean of the members' probabilities (sigmoid of each logit), not of their logits.
176        # no_grad because this is inference only.
177        with torch.no_grad():
178            p = torch.stack([torch.sigmoid(m(cat, num)) for m in self.models]).mean(dim=0).numpy()
179        # One dict or Series gives a float; a list or DataFrame gives one value per row.
180        return float(p[0]) if isinstance(scenarios, (dict, pd.Series)) else p
181
182    def out_of_range(self, scenario):
183        """List inputs outside the range seen in training, where predictions are extrapolation.
184
185        Values are not validated here (`predict` does that), so call this on a
186        scenario that `predict` accepts, as `main` does.
187
188        Parameters
189        ----------
190        scenario : dict or pandas.Series
191            One scenario (missing fields take default values, see `complete`).
192            Anything else `complete` accepts also works, but only its first
193            row is checked.
194
195        Returns
196        -------
197        list of str
198            One message per input outside the training data, empty if none:
199            a numeric input more than 2 % of its range outside the training
200            minimum-maximum; a run length or track temperature more than 2 %
201            outside the range trained for that segment (straight or any
202            corner) or weather (dry, or damp and wet); or Overtake Mode
203            inconsistent with the gap: on with a gap more than three standard
204            deviations of the detection-point noise above 1 s (1.6 s), or off
205            with a gap more than three below it (0.4 s).
206
207        Raises
208        ------
209        ValueError
210            If a field name is unknown (see `complete`) or a numeric field
211            cannot be read as a number.
212        """
213        df = complete(scenario)
214        # Only the first row is checked; see the docstring.
215        row = df.iloc[0]
216        warnings = []
217        # Run length and temperature were trained on narrower, situation-specific ranges.
218        situational = {
219            "run_length_m": ("straight" if row["segment"] == "straight" else "corner", TRAINED_RUN_LENGTH_M,
220                             "segments"),
221            "track_temp_c": ("dry" if row["track_condition"] == "dry" else "wet", TRAINED_TRACK_TEMP_C,
222                             "conditions"),
223        }
224        for col in NUMERIC:
225            value = float(row[col])
226            if col in situational:
227                key, ranges, what = situational[col]
228                lo, hi = ranges[key]
229                where = f" for {key} {what}" if key != "wet" else " for damp or wet conditions"
230            else:
231                lo, hi, where = self.preprocessor.mins[col], self.preprocessor.maxs[col], ""
232            # Small tolerance (2 % of the range) so values at the edge of the training data are not
233            # flagged as extrapolation.
234            slack = 0.02 * (hi - lo)
235            if not lo - slack <= value <= hi + slack:
236                warnings.append(f"{col}={value:g} is outside the training range {lo:g}-{hi:g}{where}")
237        # In training, Overtake Mode follows the detection-point gap, which differs from the attack gap
238        # by N(0, DETECTION_GAP_SD_S = 0.2 s). Beyond 3 SD from 1 s a mismatch has probability below
239        # 0.14 % per attempt: of 300,000 simulated attempts (seed 0), 6 have the mode on above 1.6 s
240        # (4 in the training split) and 13 have it off below 0.4 s, so the network has almost no data there.
241        gap, limit = float(row["gap_s"]), 3 * DETECTION_GAP_SD_S
242        if bool(row["overtake_mode"]) and gap > 1.0 + limit:
243            warnings.append(f"overtake_mode is on with gap_s={gap:g}; training data has it only within {1 + limit:g} s")
244        if not bool(row["overtake_mode"]) and gap < 1.0 - limit:
245            warnings.append(f"overtake_mode is off with gap_s={gap:g}; training data always has it on below {1 - limit:g} s")
246        return warnings

249def complete(scenarios):
250    """Return the scenarios as a DataFrame with every field of the schema.
251
252    Omitted fields take their defaults: `overtake.features.DEFAULTS`, except
253    ``run_length_m`` and ``track_temp_c`` (`overtake.features.situational_defaults`)
254    and ``overtake_mode`` (on when ``gap_s`` is 1 s or less). For dict, Series
255    or list-of-dict input a field set to ``None`` also means "use the default";
256    in a DataFrame, empty cells (None or NaN) are kept. Values are not checked
257    here: `overtake.features.validate` later rejects empty cells and other
258    invalid values with a message naming the field.
259
260    Parameters
261    ----------
262    scenarios : dict, pandas.Series, list of dict or pandas.DataFrame
263        One scenario (dict or Series) or several (list of dicts, or a
264        DataFrame with one row per scenario) in the `overtake.features`
265        schema. A DataFrame is copied, not modified.
266
267    Returns
268    -------
269    pandas.DataFrame
270        One row per scenario.
271
272    Raises
273    ------
274    ValueError
275        If a dict or Series has a field that is not in the schema (usually a
276        typo). Extra DataFrame columns, such as a label, are ignored.
277    """
278    if isinstance(scenarios, pd.DataFrame):
279        return _fill_columns(scenarios.reset_index(drop=True).copy())
280    rows = [dict(scenarios)] if isinstance(scenarios, (dict, pd.Series)) else [dict(r) for r in scenarios]
281    unknown = sorted({key for row in rows for key in row} - set(FEATURES))
282    if unknown:
283        raise ValueError(f"unknown field(s) {unknown}; allowed fields: {FEATURES}")
284    return pd.DataFrame([_fill_row(row) for row in rows], columns=FEATURES)

287def _fill_row(row):
288    """Fill one scenario's omitted or ``None`` fields with their defaults.
289
290    Parameters
291    ----------
292    row : dict
293        Field name -> value; names already checked by `complete`.
294
295    Returns
296    -------
297    dict
298        Every schema field: the given values, the situational run length and
299        track temperature for the given segment and weather, and
300        ``overtake_mode`` from the gap (on at 1 s or less) unless given.
301    """
302    given = {key: value for key, value in row.items() if value is not None}
303    segment = given.get("segment", DEFAULTS["segment"])
304    condition = given.get("track_condition", DEFAULTS["track_condition"])
305    filled = {**DEFAULTS, **situational_defaults(segment, condition), **given}
306    if "overtake_mode" not in given:
307        # A gap that is not a number gives False here; validate() then rejects the gap itself.
308        gap = pd.to_numeric(pd.Series([filled["gap_s"]]), errors="coerce").iloc[0]
309        filled["overtake_mode"] = bool(gap <= 1.0)
310    return filled

313def _fill_columns(df):
314    """Add any missing schema columns to a DataFrame, in place.
315
316    Columns already present are left untouched, including empty cells
317    (`overtake.features.validate` rejects those later).
318
319    Parameters
320    ----------
321    df : pandas.DataFrame
322        Scenarios, one per row; modified in place (`complete` passes a copy).
323
324    Returns
325    -------
326    pandas.DataFrame
327        The same object, with every `overtake.features.FEATURES` column.
328    """
329    for col, value in DEFAULTS.items():
330        if col in df.columns or col == "overtake_mode":
331            continue
332        if col in ("run_length_m", "track_temp_c"):
333            # DEFAULTS lists segment and track_condition first, so both columns exist by now.
334            value = np.array([situational_defaults(seg, cond)[col]
335                              for seg, cond in zip(df["segment"], df["track_condition"])], dtype=float)
336        df[col] = value
337    if "overtake_mode" not in df.columns:
338        df["overtake_mode"] = pd.to_numeric(df["gap_s"], errors="coerce") <= 1.0
339    return df

342def finite_float(text):
343    """argparse type for a finite number: ``float()`` alone would accept "nan" and "inf".
344
345    Parameters
346    ----------
347    text : str
348        The option's value as typed on the command line.
349
350    Returns
351    -------
352    float
353        The parsed value.
354
355    Raises
356    ------
357    argparse.ArgumentTypeError
358        If ``text`` is not a finite number (argparse then exits with status 2).
359    """
360    try:
361        value = float(text)
362    except ValueError:
363        raise argparse.ArgumentTypeError(f"{text!r} is not a number") from None
364    if not math.isfinite(value):
365        raise argparse.ArgumentTypeError(f"{text!r} is not a finite number")
366    return value

369def parse_args(argv=None):
370    """Parse command-line options; every option defaults to the default scenario.
371
372    ``--run-length`` and ``--track-temp`` are left as ``None`` when not given,
373    so that `complete` fills the situational value for the segment and
374    weather (see `scenario_from_args`).
375
376    Parameters
377    ----------
378    argv : list of str, optional
379        Arguments without the program name; ``sys.argv[1:]`` if omitted.
380
381    Returns
382    -------
383    argparse.Namespace
384        One attribute per option (``args.gap``, ``args.defender_battery``, ...).
385
386    Raises
387    ------
388    SystemExit
389        Status 2 for an unknown option or an invalid value (argparse prints
390        the usage and the error); status 0 after printing ``--help``.
391    """
392    p = argparse.ArgumentParser(description=__doc__.split("\n")[0])
393    p.add_argument("--condition", choices=CONDITIONS, default=DEFAULTS["track_condition"])
394    p.add_argument("--segment", choices=SEGMENTS, default=DEFAULTS["segment"])
395    p.add_argument("--circuit", choices=CIRCUITS, default=DEFAULTS["circuit"])
396    p.add_argument("--gap", type=finite_float, default=DEFAULTS["gap_s"], help="seconds to the car ahead")
397    p.add_argument("--run-length", type=finite_float,
398                   help="metres of straight before the braking point (default 800 on a straight, 150 into a corner)")
399    p.add_argument("--track-temp", type=finite_float, help="deg C (default 35 dry, 20 damp/wet)")
400
401    a = p.add_argument_group("attacker")
402    a.add_argument("--tyre", choices=COMPOUNDS, default=DEFAULTS["attacker_tyre"])
403    a.add_argument("--tyre-age", type=finite_float, default=DEFAULTS["attacker_tyre_age"], help="laps")
404    a.add_argument("--battery", type=finite_float, default=DEFAULTS["attacker_battery"],
405                   help="state of charge, %% of the 4 MJ usable energy")
406    a.add_argument("--skill", type=finite_float, default=DEFAULTS["attacker_skill"],
407                   help="racecraft rating 0-100 (grid range about 60-100)")
408    a.add_argument("--no-boost", action="store_true", help="attack without Boost")
409    a.add_argument("--overtake-mode", choices=["auto", "on", "off"], default="auto",
410                   help="auto = on when the gap is 1 s or less")
411    a.add_argument("--top-speed-delta", type=finite_float, default=0.0, help="car straight-line advantage, km/h")
412    a.add_argument("--pace-delta", type=finite_float, default=0.0, help="car lap-time advantage, s/lap")
413
414    d = p.add_argument_group("defender")
415    d.add_argument("--defender-tyre", choices=COMPOUNDS, default=DEFAULTS["defender_tyre"])
416    d.add_argument("--defender-tyre-age", type=finite_float, default=DEFAULTS["defender_tyre_age"])
417    d.add_argument("--defender-battery", type=finite_float, default=DEFAULTS["defender_battery"])
418    d.add_argument("--defender-skill", type=finite_float, default=DEFAULTS["defender_skill"])
419    d.add_argument("--defender-no-boost", action="store_true", help="defender does not use Boost")
420
421    p.add_argument("--model", type=Path, default=DEFAULT_MODEL)
422    return p.parse_args(argv)

425def scenario_from_args(args):
426    """Translate parsed options into one scenario dict.
427
428    Options left unset (run length, track temperature) are omitted so that
429    `complete` gives them their situational defaults, exactly as in Python.
430
431    Parameters
432    ----------
433    args : argparse.Namespace
434        From `parse_args`.
435
436    Returns
437    -------
438    dict
439        Schema field -> value. Every field is present except ``run_length_m``
440        and ``track_temp_c`` when their options were not given.
441    """
442    scenario = {
443        "track_condition": args.condition,
444        "segment": args.segment,
445        "circuit": args.circuit,
446        "attacker_tyre": args.tyre,
447        "defender_tyre": args.defender_tyre,
448        "gap_s": args.gap,
449        "run_length_m": args.run_length,
450        "attacker_battery": args.battery,
451        "defender_battery": args.defender_battery,
452        "attacker_tyre_age": args.tyre_age,
453        "defender_tyre_age": args.defender_tyre_age,
454        "track_temp_c": args.track_temp,
455        "attacker_skill": args.skill,
456        "defender_skill": args.defender_skill,
457        "top_speed_delta_kph": args.top_speed_delta,
458        "pace_delta_s": args.pace_delta,
459        "attacker_boost": not args.no_boost,
460        "defender_boost": not args.defender_no_boost,
461        # auto applies the 1 s rule, taking the detection-point gap equal to the attack gap.
462        "overtake_mode": {"auto": args.gap <= 1.0, "on": True, "off": False}[args.overtake_mode],
463    }
464    # Unset options are None; dropping them lets `complete` fill the situational defaults.
465    return {key: value for key, value in scenario.items() if value is not None}

468def describe(s):
469    """Format a completed scenario as the indented summary the command line prints.
470
471    Parameters
472    ----------
473    s : dict
474        A scenario with every field (see `complete`).
475
476    Returns
477    -------
478    str
479        Four indented lines (track, attacker, defender, gap) joined by
480        newlines, as printed by `main`.
481    """
482    on = {True: "on", False: "off"}
483    return "\n".join([
484        f"  Track:    {s['track_condition']}, {s['track_temp_c']:g} C, {s['circuit']}, "
485        f"{s['segment'].replace('_', ' ')} ({s['run_length_m']:g} m run)",
486        f"  Attacker: {s['attacker_tyre']} ({s['attacker_tyre_age']:g} laps), battery {s['attacker_battery']:g}%, "
487        f"Boost {on[s['attacker_boost']]}, Overtake Mode {on[s['overtake_mode']]}, skill {s['attacker_skill']:g}",
488        f"  Defender: {s['defender_tyre']} ({s['defender_tyre_age']:g} laps), battery {s['defender_battery']:g}%, "
489        f"Boost {on[s['defender_boost']]}, skill {s['defender_skill']:g}",
490        f"  Gap:      {s['gap_s']:.2f} s",
491    ])

494def main(argv=None):
495    """Command-line entry point: print P(overtake) for one scenario.
496
497    Prints the completed scenario, its probability, the change Boost makes
498    when the attacker uses it, any extrapolation warnings from
499    `OvertakePredictor.out_of_range`, and the same situation at gaps of
500    0.25-1.5 s. See ``python -m overtake.predict --help`` for the options.
501
502    Parameters
503    ----------
504    argv : list of str, optional
505        Command-line arguments without the program name; ``sys.argv[1:]`` if
506        omitted.
507
508    Raises
509    ------
510    SystemExit
511        Status 1 if there is no trained model at ``--model``; status 2 if an
512        option is malformed (argparse) or the scenario fails
513        `overtake.features.validate`.
514
515    Examples
516    --------
517    With the committed model this prints ``P(overtake) = 19.7%``::
518
519        python -m overtake.predict --segment slow_corner --gap 0.6 --battery 70
520    """
521    args = parse_args(argv)
522    try:
523        predictor = OvertakePredictor.load(args.model)
524    except FileNotFoundError as error:
525        print(error, file=sys.stderr)
526        raise SystemExit(1)
527    try:
528        # Fill the situational defaults once, so that what is printed is what is predicted.
529        scenario = complete(scenario_from_args(args)).iloc[0].to_dict()
530        p = predictor.predict(scenario)
531    except ValueError as error:
532        print(f"invalid input: {error}", file=sys.stderr)
533        raise SystemExit(2)
534
535    print("Scenario")
536    print(describe(scenario))
537    print(f"\nP(overtake) = {p:.1%}")
538    if scenario["attacker_boost"]:
539        # Boost effect: the same scenario with only the attacker's Boost switched off.
540        p_no_boost = predictor.predict({**scenario, "attacker_boost": False})
541        print(f"  without Boost: {p_no_boost:.1%}  (Boost adds {100 * (p - p_no_boost):+.1f} points)")
542    for warning in predictor.out_of_range(scenario):
543        print(f"  warning: {warning}")
544
545    print("\nSame situation at other gaps")
546    # Vary only the gap. With --overtake-mode auto the mode follows each gap by the 1 s rule; a forced
547    # on or off is held fixed.
548    for gap in (0.25, 0.5, 0.75, 1.0, 1.25, 1.5):
549        q = predictor.predict({**scenario, "gap_s": gap,
550                               "overtake_mode": gap <= 1.0 if args.overtake_mode == "auto" else scenario["overtake_mode"]})
