overtake.realdata

Real 2026 race data from the OpenF1 API, for comparison with the simulator.

python -m overtake.realdata download      # fetch and cache (about 4 minutes, once)

OpenF1 (https://openf1.org) is a free, unofficial API of Formula 1 timing data. No public source records battery charge or Boost / Overtake Mode use (the old DRS channel is empty in 2026), so the real data can test the simulator only indirectly: speeds, overtakes and gaps.

Responses are cached as JSON under data/real/openf1/ (not in git), so the download runs once and every later analysis is offline and repeatable. OpenF1's free tier allows about 30 requests a minute, so requests are spaced 2.2 s apart.

 46def fetch(endpoint, **params):
 47    """Return an OpenF1 response, from the cache if present, otherwise from the API.
 48
 49    Parameters
 50    ----------
 51    endpoint : str
 52        OpenF1 endpoint name, e.g. ``"laps"``.
 53    **params
 54        Query parameters, e.g. ``session_key=11361``.
 55
 56    Returns
 57    -------
 58    list of dict
 59
 60    Raises
 61    ------
 62    urllib.error.URLError
 63        If the data is not cached and the API cannot be reached.
 64    """
 65    global _last_request
 66    name = endpoint + "".join(f"_{k}-{v}" for k, v in sorted(params.items())) + ".json"
 67    path = CACHE / name
 68    if path.exists():
 69        return json.loads(path.read_text())
 70    wait = REQUEST_SPACING_S - (time.monotonic() - _last_request)
 71    if wait > 0:
 72        time.sleep(wait)
 73    url = f"{API}/{endpoint}?{urllib.parse.urlencode(params)}"
 74    for attempt in range(4):
 75        try:
 76            with urllib.request.urlopen(url, timeout=120) as response:
 77                data = json.loads(response.read())
 78            break
 79        except urllib.error.HTTPError as error:
 80            if error.code == 404:  # OpenF1 answers 404 "No results found" when there is no data
 81                data = []
 82                break
 83            if error.code != 429 or attempt == 3:  # 429 = rate limited: wait and retry
 84                raise
 85            time.sleep(15 * (attempt + 1))
 86    _last_request = time.monotonic()
 87    CACHE.mkdir(parents=True, exist_ok=True)
 88    path.write_text(json.dumps(data))
 89    return data

 92def completed_races(year=2026, before=RACES_BEFORE):
 93    """Grand Prix races (not sprints) of ``year`` dated before ``before`` (YYYY-MM-DD).
 94
 95    Some may have no data in OpenF1; `download` skips those.
 96
 97    Returns
 98    -------
 99    list of dict
100        OpenF1 session records, in calendar order.
101    """
102    sessions = fetch("sessions", year=year, session_type="Race")
103    return [s for s in sessions if s["session_name"] == "Race" and s["date_start"][:10] < before]

106def download(log=print):
107    """Fetch and cache everything the comparison needs.
108
109    Returns
110    -------
111    list of dict
112        The races downloaded.
113    """
114    races = completed_races()
115    log(f"{len(races)} completed races")
116    for race in races:
117        if not fetch("laps", session_key=race["session_key"]):
118            log(f"  {race['circuit_short_name']:18s} no data in OpenF1, skipped")
119            continue
120        counts = []
121        for endpoint in RACE_ENDPOINTS:
122            counts.append(f"{endpoint} {len(fetch(endpoint, session_key=race['session_key']))}")
123        log(f"  {race['circuit_short_name']:18s} " + ", ".join(counts))
124    monza = next(r for r in races if r["circuit_short_name"] == TELEMETRY["session"])
125    rows = fetch("car_data", session_key=monza["session_key"], driver_number=TELEMETRY["driver_number"])
126    log(f"  telemetry: car {TELEMETRY['driver_number']} at Monza, {len(rows):,} samples")
127    return races

150def _times(values):
151    """Parse OpenF1 timestamps to timezone-naive UTC (so every comparison uses one convention)."""
152    return pd.to_datetime(pd.Series(values), utc=True, format="ISO8601").dt.tz_localize(None)

155def lap_index(laps):
156    """Start time of every lap of every driver, for looking up which lap a moment falls in.
157
158    Parameters
159    ----------
160    laps : list of dict
161        OpenF1 ``laps`` rows.
162
163    Returns
164    -------
165    dict of int -> pandas.DataFrame
166        Per driver: ``lap_number``, ``start`` (UTC), ``pit_out`` (bool), sorted by
167        lap. Lap 1 has no recorded start (standing start) and is left out.
168    """
169    df = pd.DataFrame(laps)
170    df = df[df["date_start"].notna()].copy()
171    df["start"] = _times(df["date_start"]).values
172    out = {}
173    for driver, rows in df.groupby("driver_number"):
174        out[int(driver)] = rows.sort_values("lap_number")[["lap_number", "start", "is_pit_out_lap"]].rename(
175            columns={"is_pit_out_lap": "pit_out"}).reset_index(drop=True)
176    return out

179def lap_at(index, driver, when):
180    """Lap a driver was on at time ``when`` (1 if before their lap-2 start, None if unknown)."""
181    rows = index.get(int(driver))
182    if rows is None or rows.empty:
183        return None
184    i = rows["start"].searchsorted(when, side="right") - 1
185    return 1 if i < 0 else int(rows["lap_number"].iloc[i])

188# Race-control messages that start a period without normal racing, and what kind it is.
189NEUTRALISING = {"OVERTAKE DISABLED": "overtake mode", "SAFETY CAR DEPLOYED": "safety car",
190                "VSC DEPLOYED": "safety car", "RED FLAG - RACE SUSPENDED": "red flag"}

193def neutralised_periods(race_control):
194    """Time intervals without normal racing with Overtake Mode available, from race-control messages.
195
196    A period starts with OVERTAKE DISABLED, SAFETY CAR DEPLOYED, VSC DEPLOYED
197    or RED FLAG and lasts until every such cause has ended: OVERTAKE ENABLED
198    ends all of them; TRACK CLEAR also ends a (virtual) safety car, but not a
199    red flag, after which the race restarts from the grid. Both kinds of
200    message are needed because race control does not announce every safety
201    car with OVERTAKE DISABLED.
202
203    Parameters
204    ----------
205    race_control : list of dict
206        OpenF1 ``race_control`` rows for one race.
207
208    Returns
209    -------
210    list of (pandas.Timestamp, pandas.Timestamp)
211        Start and end (UTC); the end is ``pandas.Timestamp.max`` if the race
212        finished neutralised.
213    """
214    rows = sorted(race_control, key=lambda m: m["date"])
215    times = _times([m["date"] for m in rows])
216    active, periods, start = set(), [], None
217    for m, when in zip(rows, times):
218        message = m.get("message") or ""
219        if message in NEUTRALISING:
220            active.add(NEUTRALISING[message])
221        elif message == "OVERTAKE ENABLED":
222            active.clear()
223        elif message == "TRACK CLEAR":
224            active.discard("safety car")
225        if active and start is None:
226            start = when
227        elif not active and start is not None:
228            periods.append((start, when))
229            start = None
230    if start is not None:
231        periods.append((start, pd.Timestamp.max))
232    return periods

235def _pit_laps(pits):
236    out = {}
237    for p in pits:
238        if p.get("lap_number") is not None:
239            out.setdefault(int(p["driver_number"]), set()).add(int(p["lap_number"]))
240    return out

243def classify_overtakes(overtakes, laps, pits, race_control=()):
244    """Label every position change with why it happened.
245
246    OpenF1's overtakes are position changes between two cars. Each gets the
247    first reason that applies:
248
249    * ``"start"``: lap 1, where positions shuffle from a standing start;
250    * ``"neutralised"``: under a safety car, virtual safety car or red flag, or
251      while race control had disabled Overtake Mode (`neutralised_periods`);
252    * ``"pit stop"``: either car pitted on that lap or the lap before, or is
253      on a pit-out lap (one car stopping for tyres is not an overtake);
254    * ``"on track"``: everything else.
255
256    Not identifiable from the data, so left as on-track passes: passes on cars
257    slowing with a problem.
258
259    Parameters
260    ----------
261    overtakes, laps, pits, race_control : list of dict
262        OpenF1 rows for one race.
263
264    Returns
265    -------
266    pandas.DataFrame
267        The overtakes with ``time``, ``lap`` and ``reason`` columns added.
268    """
269    df = pd.DataFrame(overtakes)
270    if df.empty:
271        return df.assign(time=pd.Series(dtype="datetime64[ns]"), lap=pd.Series(dtype=int),
272                         reason=pd.Series(dtype=str))
273    index, pit_laps = lap_index(laps), _pit_laps(pits)
274    periods = neutralised_periods(race_control)
275    df["time"] = _times(df["date"]).values
276    df["lap"] = [lap_at(index, d, t) for d, t in zip(df["overtaking_driver_number"], df["time"])]
277
278    def pit_related(driver, lap):
279        rows = index.get(int(driver))
280        out_lap = rows is not None and bool(rows.loc[rows["lap_number"] == lap, "pit_out"].any())
281        return out_lap or bool(pit_laps.get(int(driver), set()) & {lap, lap - 1})
282
283    def reason(a, b, lap, when):
284        if lap is None or lap <= 1:
285            return "start"
286        if any(lo <= when < hi for lo, hi in periods):
287            return "neutralised"
288        if pit_related(a, lap) or pit_related(b, lap):
289            return "pit stop"
290        return "on track"
291
292    df["reason"] = [reason(a, b, lap, t) for a, b, lap, t in
293                    zip(df["overtaking_driver_number"], df["overtaken_driver_number"], df["lap"], df["time"])]
294    return df

297def on_track_overtakes(overtakes, laps, pits, race_control=()):
298    """Only the position changes that look like real on-track passes (see `classify_overtakes`).

309def wilson_interval(successes, n, z=1.96):
310    """95 % Wilson score interval for a proportion (well behaved for small counts).
311
312    Returns
313    -------
314    (low, high) : tuple of float
315    """
316    if n == 0:
317        return (np.nan, np.nan)
318    p = successes / n
319    centre = (p + z * z / (2 * n)) / (1 + z * z / n)
320    half = z * np.sqrt(p * (1 - p) / n + z * z / (4 * n * n)) / (1 + z * z / n)
321    return (centre - half, centre + half)

324def laps_by_gap(laps, intervals, pits, passes, race_control=()):
325    """One row per driver-lap: gap to the car ahead at the start of the lap, and whether the driver passed.
326
327    The gap is the ``interval`` sample nearest the lap's start (within 10 s);
328    the lap counts as a pass if the driver made an on-track overtake before
329    their next lap began. Lap 1, pit in/out laps, laps that overlap a
330    neutralised period (`neutralised_periods`), the race leader and lapped
331    gaps ("+1 LAP") are left out.
332
333    Returns
334    -------
335    pandas.DataFrame
336        Columns ``driver``, ``lap``, ``gap_s``, ``passed`` (bool).
337    """
338    index = lap_index(laps)
339    iv = pd.DataFrame(intervals)
340    iv["gap"] = pd.to_numeric(iv["interval"], errors="coerce")
341    iv = iv[iv["gap"].notna() & (iv["gap"] > 0)].copy()
342    iv["time"] = _times(iv["date"]).values
343    pit_laps, periods = _pit_laps(pits), neutralised_periods(race_control)
344    pass_times = {}
345    for d, t in zip(passes.get("overtaking_driver_number", []), passes.get("time", [])):
346        pass_times.setdefault(int(d), []).append(t)
347
348    rows = []
349    for driver, lap_rows in index.items():
350        samples = iv[iv["driver_number"] == driver].sort_values("time")
351        if samples.empty:
352            continue
353        times = samples["time"].to_numpy()
354        mine = np.array(sorted(pass_times.get(driver, [])), dtype="datetime64[ns]")
355        for i in range(len(lap_rows) - 1):
356            lap, start, end = int(lap_rows["lap_number"][i]), lap_rows["start"][i], lap_rows["start"][i + 1]
357            if lap <= 1 or lap_rows["pit_out"][i] or lap in pit_laps.get(driver, set()):
358                continue
359            if any(lo < end and start < hi for lo, hi in periods):
360                continue
361            j = np.searchsorted(times, start.to_datetime64())
362            candidates = [k for k in (j - 1, j) if 0 <= k < len(times)]
363            k = min(candidates, key=lambda k: abs(times[k] - start.to_datetime64()))
364            if abs(times[k] - start.to_datetime64()) > np.timedelta64(10, "s"):
365                continue
366            passed = bool(((mine >= start.to_datetime64()) & (mine < end.to_datetime64())).any())
367            rows.append((driver, lap, float(samples["gap"].iloc[k]), passed))
368    return pd.DataFrame(rows, columns=["driver", "lap", "gap_s", "passed"])

371def rate_by_gap(gap_s, passed, bins=GAP_BINS):
372    """Pass rate (with 95 % Wilson interval) in each gap bin.
373
374    Returns
375    -------
376    list of dict
377        ``bin`` label, ``low``/``high`` edges, ``n``, ``passes``, ``rate``, ``ci``.
378    """
379    gap_s, passed = np.asarray(gap_s, dtype=float), np.asarray(passed, dtype=bool)
380    out = []
381    for lo, hi in zip(bins[:-1], bins[1:]):
382        mask = (gap_s >= lo) & (gap_s < hi)
383        n, k = int(mask.sum()), int(passed[mask].sum())
384        out.append({"bin": f"{lo:g}-{hi:g}", "low": lo, "high": hi, "n": n, "passes": k,
385                    "rate": k / n if n else None, "ci": list(wilson_interval(k, n)) if n else [None, None]})
386    return out

389def full_throttle_runs(car_data, min_seconds=6.0, min_gain_kph=80):
390    """Stretches of telemetry where the car accelerates flat out (straights).
391
392    A run is a sequence of samples with throttle >= 99 % and no brake, lasting
393    at least ``min_seconds`` and gaining at least ``min_gain_kph``. Distance
394    is the trapezoidal integral of speed over time.
395
396    Returns
397    -------
398    list of pandas.DataFrame
399        Per run: ``t`` (s from run start), ``x`` (m), ``kph``.
400    """
401    df = pd.DataFrame(car_data)
402    df["time"] = _times(df["date"]).values
403    df = df.sort_values("time").reset_index(drop=True)
404    flat = (df["throttle"] >= 99) & (df["brake"] == 0)
405    runs, start = [], None
406    for i, on in enumerate(list(flat) + [False]):
407        if on and start is None:
408            start = i
409        elif not on and start is not None:
410            seg = df.iloc[start:i]
411            t = (seg["time"] - seg["time"].iloc[0]).dt.total_seconds().to_numpy()
412            v = seg["speed"].to_numpy(dtype=float) / 3.6
413            if t[-1] >= min_seconds and (v.max() - v[0]) * 3.6 >= min_gain_kph:
414                x = np.concatenate([[0.0], np.cumsum(0.5 * (v[1:] + v[:-1]) * np.diff(t))])
415                runs.append(pd.DataFrame({"t": t, "x": x, "kph": v * 3.6}))
416            start = None
417    return runs

420def simulate_run(v0_kph, duration_s, deploy_share, overtake_mode=False, battery=1.0, dt=0.05):
421    """Speed of one car accelerating from ``v0_kph`` for ``duration_s``, from the simulator's physics.
422
423    Uses the same force model and RK4 integrator as the simulator (new soft
424    tyre, dry, no slipstream). ``deploy_share`` is 1 with Boost, 0.55 on the
425    standard map, 0 with no electrical power.
426
427    Returns
428    -------
429    pandas.DataFrame
430        ``t`` (s), ``x`` (m), ``kph``.
431    """
432    car = physics.CarParameters()
433    state = physics.CarState(mu=np.array([car.mu_reference]), cda_m2=np.array([car.cda_m2]),
434                             deploy_share=np.array([float(deploy_share)]),
435                             energy_j=np.array([physics.BATTERY_USABLE_J * battery
436                                                + physics.OVERTAKE_MODE_EXTRA_J * overtake_mode]),
437                             overtake_mode=np.array([overtake_mode]))
438
439    def f(y):
440        accel, ers = physics.acceleration(y[1:2], y[2:3], state, state.cda_m2, car)
441        return np.array([y[1], accel[0], ers[0]])
442
443    y, rows = np.array([0.0, v0_kph / 3.6, 0.0]), []
444    for i in range(int(round(duration_s / dt)) + 1):
445        rows.append((i * dt, y[0], y[1] * 3.6))
446        y = physics.rk4_step(f, y, dt)
447    return pd.DataFrame(rows, columns=["t", "x", "kph"])

450def analyse(log=print):
451    """Run every comparison on the cached data and return the results.
452
453    Steps:
454
455    1. Classify each race's position changes (start, neutralised, pit stop,
456       on track) and keep the on-track passes.
457    2. Pass rate per lap against the gap at the start of the lap, real and
458       simulated (the simulator's mean true probability per attempt over its
459       own scenario distribution). Both are also expressed relative to the
460       0.25-0.5 s bin, because a per-lap rate and a per-attempt probability
461       differ in level but can be compared in shape.
462    3. Real on-track passes per race lap against the simulator's circuit
463       passability (Spearman rank correlation).
464    4. Speed-trap readings per race against the 2026 MGU-K limit (355 km/h).
465    5. Full-throttle stretches of one car's telemetry against the RK4 car
466       model started from the same speed, with full Boost, the standard map
467       and no electric power.
468
469    Returns
470    -------
471    dict
472    """
473    from scipy.stats import spearmanr
474
475    from . import simulator
476
477    races, gap_rows, speed_rows = [], [], []
478    for race in completed_races():
479        key = race["session_key"]
480        laps = fetch("laps", session_key=key)
481        if not laps:
482            continue
483        overtakes, pits = fetch("overtakes", session_key=key), fetch("pit", session_key=key)
484        control = fetch("race_control", session_key=key)
485        labelled = classify_overtakes(overtakes, laps, pits, control)
486        passes = labelled[labelled["reason"] == "on track"]
487        counts = labelled["reason"].value_counts()
488        race_laps = max(l["lap_number"] for l in laps)
489        races.append({"race": race["circuit_short_name"], "circuit": CIRCUIT_MAP.get(race["circuit_short_name"]),
490                      "position_changes": len(overtakes), "start": int(counts.get("start", 0)),
491                      "neutralised": int(counts.get("neutralised", 0)), "pit_stop": int(counts.get("pit stop", 0)),
492                      "on_track": len(passes), "race_laps": race_laps, "on_track_per_lap": len(passes) / race_laps})
493        table = laps_by_gap(laps, fetch("intervals", session_key=key), pits, passes, control)
494        gap_rows.append(table)
495        top = pd.DataFrame(laps)[["i1_speed", "i2_speed", "st_speed"]].max(axis=1).dropna()
496        speed_rows.append({"race": race["circuit_short_name"], "laps": int(len(top)), "median": float(top.median()),
497                           "p99": float(top.quantile(0.99)), "max": float(top.max()),
498                           "share_over_355": float((top > 355).mean())})
499        log(f"  {race['circuit_short_name']:18s} {len(passes):4d} on-track passes, {len(table):5d} driver-laps")
500
501    # Gap: real per-lap rate vs the simulator's mean probability per attempt.
502    gaps = pd.concat(gap_rows, ignore_index=True)
503    real_gap = rate_by_gap(gaps["gap_s"], gaps["passed"])
504    sim = simulator.generate_dataset(300_000, seed=0)
505    sim_gap = []
506    for lo, hi in zip(GAP_BINS[:-1], GAP_BINS[1:]):
507        mask = (sim["gap_s"] >= lo) & (sim["gap_s"] < hi)

391    """Stretches of telemetry where the car accelerates flat out (straights).
392
393    A run is a sequence of samples with throttle >= 99 % and no brake, lasting
394    at least ``min_seconds`` and gaining at least ``min_gain_kph``. Distance
395    is the trapezoidal integral of speed over time.
396
397    Returns
398    -------
399    list of pandas.DataFrame
400        Per run: ``t`` (s from run start), ``x`` (m), ``kph``.
401    """

405    flat = (df["throttle"] >= 99) & (df["brake"] == 0)

408        if on and start is None:
409            start = i
410        elif not on and start is not None:
411            seg = df.iloc[start:i]
412            t = (seg["time"] - seg["time"].iloc[0]).dt.total_seconds().to_numpy()
