overtake.export

Export every simulated attempt and every experiment result to one Excel workbook.

python -m overtake.export                         # writes data/overtake_simulations.xlsx
python -m overtake.export --out my_copy.xlsx

139def attempts_table(n=300_000, seed=SEED, log=print):
140    """Every simulated attempt with its physics results, outcome and predictions.
141
142    Parameters
143    ----------
144    n : int, optional
145        Number of attempts (300,000 is the dataset used throughout the project).
146    seed : int, optional
147    log : callable, optional
148
149    Returns
150    -------
151    pandas.DataFrame
152        One row per attempt, columns as listed in `ATTEMPT_COLUMNS`.
153    """
154    log(f"simulating {n:,} attempts")
155    df = simulator.generate_dataset(n, seed=seed)
156    train_df, val_df, test_df = split(df, seed)
157    labels = pd.Series("train", index=df.index)
158    labels[val_df.index] = "validation"
159    labels[test_df.index] = "test"
160
161    log("recording physics results")
162    parts = []
163    for start in range(0, n, 20_000):
164        part = df.iloc[start:start + 20_000]
165        run, grip_a, grip_d = simulator.closing_run(part)
166        deficit, sigma, z0 = simulator.outcome_parameters(part, run, grip_a, grip_d)
167        parts.append(pd.DataFrame({
168            "attacker_grip": grip_a, "defender_grip": grip_d,
169            "deficit_at_braking_point_s": run.deficit_s,
170            "attacker_speed_kph": run.attacker_speed * 3.6, "defender_speed_kph": run.defender_speed * 3.6,
171            "attacker_energy_used_mj": run.attacker_energy_j / 1e6,
172            "defender_energy_used_mj": run.defender_energy_j / 1e6,
173            "run_time_s": run.time_s, "effective_deficit_s": deficit, "execution_sd_s": sigma,
174            "completion_logit_z0": z0,
175        }, index=part.index))
176    details = pd.concat(parts)
177
178    log("predicting with the network and the alternatives")
179    predictions = pd.DataFrame({
180        "p_network": OvertakePredictor.load().predict(df).astype(np.float64),  # float32 would round untidily
181        "p_gradient_boosting": baselines.gradient_boosting(seed).fit(train_df).predict(df),
182        "p_logistic_regression": baselines.logistic_regression().fit(train_df).predict(df),
183    }, index=df.index)
184
185    out = pd.concat([df[list(INPUTS)], details, df[[LABEL, "p_true"]], predictions], axis=1)
186    out.insert(0, "split", labels)
187    out.insert(0, "attempt_id", np.arange(1, n + 1))
188    decimals = {"gap_s": 4, "run_length_m": 1, "attacker_battery": 2, "defender_battery": 2, "track_temp_c": 1,
189                "attacker_skill": 1, "defender_skill": 1, "top_speed_delta_kph": 2, "pace_delta_s": 3,
190                "attacker_grip": 4, "defender_grip": 4, "deficit_at_braking_point_s": 4, "attacker_speed_kph": 1,
191                "defender_speed_kph": 1, "attacker_energy_used_mj": 3, "defender_energy_used_mj": 3,
192                "run_time_s": 3, "effective_deficit_s": 4, "execution_sd_s": 3, "completion_logit_z0": 3,
193                "p_true": 5, "p_network": 5, "p_gradient_boosting": 5, "p_logistic_regression": 5}
194    out = out.round(decimals)
195    # Flags as 1/0 like the outcome (LibreOffice would turn TRUE/FALSE cells into formulas).
196    for col in ("attacker_tyre_age", "defender_tyre_age", LABEL, *BINARY):
197        out[col] = out[col].astype(int)
198    return out[[c for c, _, _ in ATTEMPT_COLUMNS]]

201class _Sheet:
202    """Small helper for appending styled rows to a write-only worksheet."""
203
204    def __init__(self, wb, name, widths):
205        """Create sheet ``name`` in workbook ``wb`` with the given column widths (characters)."""
206        self.ws = wb.create_sheet(name)
207        self.rows = 0  # rows written so far; write-only sheets do not track this
208        for i, width in enumerate(widths, start=1):
209            self.ws.column_dimensions[get_column_letter(i)].width = width
210
211    def append(self, cells):
212        """Append one row of cells and count it."""
213        self.ws.append(cells)
214        self.rows += 1
215
216    def _cell(self, value, font=None, fill=None, wrap=False, number_format=None):
217        """A write-only cell holding ``value`` with the optional font, fill, wrapping and number format."""
218        cell = WriteOnlyCell(self.ws, value=value)
219        if font:
220            cell.font = font
221        if fill:
222            cell.fill = fill
223        if wrap:
224            cell.alignment = WRAP
225        if number_format:
226            cell.number_format = number_format
227        return cell
228
229    def title(self, text):
230        """Write the sheet's title line."""
231        self.append([self._cell(text, TITLE_FONT)])
232
233    def section(self, text):
234        """Write a blank line, then a section heading."""
235        self.append([])
236        self.append([self._cell(text, SECTION_FONT)])
237
238    def note(self, text):
239        """Write an explanatory note in the muted note style."""
240        self.append([self._cell(text, NOTE_FONT)])
241
242    def header(self, names):
243        """Write a table header row (bold, shaded, wrapped)."""
244        self.append([self._cell(n, HEADER_FONT, HEADER_FILL, wrap=True) for n in names])
245
246    def row(self, values, wrap=False, formats=None):
247        """Write one data row; ``formats`` maps a column index to an Excel number format."""
248        formats = formats or {}
249        self.append([self._cell(_native(v), wrap=wrap, number_format=formats.get(i))
250                     for i, v in enumerate(values)])
251
252    def table(self, header, rows, wrap=False, formats=None):
253        """Write a header row followed by every row in ``rows``."""
254        self.header(header)
255        for r in rows:
256            self.row(r, wrap=wrap, formats=formats)

259def _native(value):
260    """Convert a NumPy scalar to the plain Python type openpyxl accepts, and a list to comma-separated text."""
261    if isinstance(value, (np.bool_,)):
262        return bool(value)
263    if isinstance(value, np.integer):
264        return int(value)
265    if isinstance(value, np.floating):
266        return float(value)
267    if isinstance(value, (list, tuple)):
268        return ", ".join(str(v) for v in value)
269    return value
